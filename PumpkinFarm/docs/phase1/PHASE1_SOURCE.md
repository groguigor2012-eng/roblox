### `ReplicatedStorage.Shared.Utilities.TableUtil`

ModuleScript · `PumpkinFarm/src/shared/Utilities/TableUtil.luau` · 106 lines

```lua
--!strict
--[[
	TableUtil
	Path: ReplicatedStorage.Shared.Utilities.TableUtil

	Table helpers shared by configs (deep freezing), the data layer (deep copy +
	template reconciliation) and gameplay (counting, shuffling).

	Dependencies: none.

	API:
		DeepCopy(value) -> value
		DeepFreeze(table) -> table          -- freezes nested tables, returns the same table
		Reconcile(target, template) -> number  -- fills missing keys from template, returns keys added
		Count(table) -> number
		Keys(table) -> { key }               -- sorted when keys are strings or numbers
		Shuffle(array, rng) -> array         -- in-place Fisher-Yates
		Sum(array, selector?) -> number
]]

local TableUtil = {}

function TableUtil.DeepCopy<T>(value: T): T
	if type(value) ~= "table" then
		return value
	end
	local copy = {}
	for key, item in value :: any do
		copy[TableUtil.DeepCopy(key)] = TableUtil.DeepCopy(item)
	end
	return copy :: any
end

function TableUtil.DeepFreeze<T>(value: T): T
	if type(value) ~= "table" or table.isfrozen(value :: any) then
		return value
	end
	for _, item in value :: any do
		if type(item) == "table" then
			TableUtil.DeepFreeze(item)
		end
	end
	table.freeze(value :: any)
	return value
end

-- Adds keys present in `template` but missing from `target`. Existing values are never
-- overwritten, even when their type differs: overwriting would destroy player data, so
-- schema changes must go through Migrations instead.
function TableUtil.Reconcile(target: { [any]: any }, template: { [any]: any }): number
	local added = 0
	for key, templateValue in template do
		local current = target[key]
		if current == nil then
			target[key] = TableUtil.DeepCopy(templateValue)
			added += 1
		elseif type(current) == "table" and type(templateValue) == "table" then
			added += TableUtil.Reconcile(current, templateValue)
		end
	end
	return added
end

function TableUtil.Count(tbl: { [any]: any }): number
	local count = 0
	for _ in tbl do
		count += 1
	end
	return count
end

function TableUtil.Keys<K>(tbl: { [K]: any }): { K }
	local keys = {}
	for key in tbl do
		table.insert(keys, key)
	end
	table.sort(keys, function(a: any, b: any)
		local ta, tb = type(a), type(b)
		if ta ~= tb then
			return ta < tb
		end
		if ta == "string" or ta == "number" then
			return a < b
		end
		return tostring(a) < tostring(b)
	end)
	return keys
end

function TableUtil.Shuffle<T>(array: { T }, rng: Random): { T }
	for i = #array, 2, -1 do
		local j = rng:NextInteger(1, i)
		array[i], array[j] = array[j], array[i]
	end
	return array
end

function TableUtil.Sum<T>(array: { T }, selector: ((T) -> number)?): number
	local total = 0
	for _, item in array do
		total += if selector then selector(item) else (item :: any)
	end
	return total
end

return TableUtil
```

### `ReplicatedStorage.Shared.Utilities.Validate`

ModuleScript · `PumpkinFarm/src/shared/Utilities/Validate.luau` · 133 lines

```lua
--!strict
--[[
	Validate
	Path: ReplicatedStorage.Shared.Utilities.Validate

	Payload validation for everything that crosses the network. Each remote in
	NetworkConfig declares an argument schema; RemoteService runs CheckArgs before a
	handler ever sees the data, so handlers can rely on types and ranges.

	Schema entry kinds:
		{ Kind = "string",  MaxLength = n, Pattern = luaPattern? }
		{ Kind = "integer", Min = n, Max = n }
		{ Kind = "number",  Min = n, Max = n }        -- always rejects NaN / inf
		{ Kind = "boolean" }
		{ Kind = "Vector3", MaxMagnitude = n }        -- rejects NaN / inf components
		{ Kind = "enum",    Values = { string } }
		Optional = true on any entry allows nil.

	Dependencies: none.

	API:
		IsFiniteNumber(value) -> boolean
		IsFiniteVector3(value) -> boolean
		CheckValue(value, rule) -> (boolean, string?)
		CheckArgs(args, argCount, schema) -> (boolean, string?)
]]

export type Rule = {
	Kind: string,
	Optional: boolean?,
	MaxLength: number?,
	Pattern: string?,
	Min: number?,
	Max: number?,
	MaxMagnitude: number?,
	Values: { string }?,
}

local Validate = {}

function Validate.IsFiniteNumber(value: any): boolean
	return type(value) == "number" and value == value and value ~= math.huge and value ~= -math.huge
end

function Validate.IsFiniteVector3(value: any): boolean
	return typeof(value) == "Vector3"
		and Validate.IsFiniteNumber(value.X)
		and Validate.IsFiniteNumber(value.Y)
		and Validate.IsFiniteNumber(value.Z)
end

local function checkRange(value: number, rule: Rule): (boolean, string?)
	if rule.Min ~= nil and value < rule.Min then
		return false, "below minimum"
	end
	if rule.Max ~= nil and value > rule.Max then
		return false, "above maximum"
	end
	return true, nil
end

function Validate.CheckValue(value: any, rule: Rule): (boolean, string?)
	if value == nil then
		if rule.Optional then
			return true, nil
		end
		return false, "missing value"
	end

	local kind = rule.Kind
	if kind == "string" then
		if type(value) ~= "string" then
			return false, "expected string"
		end
		if #value == 0 then
			return false, "empty string"
		end
		if rule.MaxLength and #value > rule.MaxLength then
			return false, "string too long"
		end
		if rule.Pattern and not string.match(value, rule.Pattern) then
			return false, "string format"
		end
		return true, nil
	elseif kind == "integer" then
		if not Validate.IsFiniteNumber(value) or value % 1 ~= 0 then
			return false, "expected integer"
		end
		return checkRange(value, rule)
	elseif kind == "number" then
		if not Validate.IsFiniteNumber(value) then
			return false, "expected finite number"
		end
		return checkRange(value, rule)
	elseif kind == "boolean" then
		if type(value) ~= "boolean" then
			return false, "expected boolean"
		end
		return true, nil
	elseif kind == "Vector3" then
		if not Validate.IsFiniteVector3(value) then
			return false, "expected finite Vector3"
		end
		if rule.MaxMagnitude and (value :: Vector3).Magnitude > rule.MaxMagnitude then
			return false, "vector too large"
		end
		return true, nil
	elseif kind == "enum" then
		if type(value) ~= "string" or rule.Values == nil or table.find(rule.Values, value) == nil then
			return false, "unknown enum value"
		end
		return true, nil
	end

	return false, `unknown rule kind {kind}`
end

-- `argCount` is select("#", ...) of the received payload, which catches extra
-- trailing arguments that a plain table length would miss.
function Validate.CheckArgs(args: { any }, argCount: number, schema: { Rule }): (boolean, string?)
	if argCount > #schema then
		return false, "too many arguments"
	end
	for index, rule in schema do
		local ok, reason = Validate.CheckValue(args[index], rule)
		if not ok then
			return false, `arg {index}: {reason}`
		end
	end
	return true, nil
end

return Validate
```

```lua
function MathUtil.FlatDistance(a: Vector3, b: Vector3): number
	local dx, dz = a.X - b.X, a.Z - b.Z
	return math.sqrt(dx * dx + dz * dz)

function MathUtil.FlatDirection(from: Vector3, to: Vector3): Vector3
	local delta = Vector3.new(to.X - from.X, 0, to.Z - from.Z)
	if delta.Magnitude < 1e-4 then
		return Vector3.zero
	end
	return delta.Unit
end

	local fa, fb = MathUtil.Flat(a), MathUtil.Flat(b)
	if fa.Magnitude < 1e-4 or fb.Magnitude < 1e-4 then
		return 0
	end
	local dot = math.clamp(fa.Unit:Dot(fb.Unit), -1, 1)
	return math.deg(math.acos(dot))

-- `flightTime` seconds under constant downward `gravity`.

	local t = math.max(flightTime, 0.05)
	local delta = to - from
	local horizontal = Vector3.new(delta.X, 0, delta.Z) / t
	local vertical = (delta.Y + 0.5 * gravity * t * t) / t
	return horizontal + Vector3.new(0, vertical, 0)
end

	return origin + velocity * t + Vector3.new(0, -0.5 * gravity * t * t, 0)

	return a + (b - a) * t
end

	if a == b then

	return math.clamp((v - a) / (b - a), 0, 1)

		return value
```

```lua
--[[
	WeightedRandom
	Path: ReplicatedStorage.Shared.Utilities.WeightedRandom

	Deterministic weighted selection driven by an explicit Random object, so the seed
	shop and wave composer produce reproducible results from a match seed (useful for
	tests and for replaying bug reports).

	Dependencies: none.

	API:
		Pick(items, weightOf, rng) -> item?                    -- nil when total weight <= 0
		PickManyUnique(items, count, weightOf, rng) -> { item } -- weighted, without replacement
]]

local WeightedRandom = {}

function WeightedRandom.Pick<T>(items: { T }, weightOf: (T) -> number, rng: Random): T?
	local total = 0
	for _, item in items do
		total += math.max(0, weightOf(item))
	end
	if total <= 0 then
		return nil
	end

	local roll = rng:NextNumber() * total
	local cumulative = 0
	for _, item in items do
		cumulative += math.max(0, weightOf(item))
		if roll < cumulative then
			return item
		end
	end
	-- Floating point guard: the roll can equal the total when NextNumber returns ~1.
	for index = #items, 1, -1 do
		if weightOf(items[index]) > 0 then
			return items[index]
		end
	end
	return nil
end

function WeightedRandom.PickManyUnique<T>(items: { T }, count: number, weightOf: (T) -> number, rng: Random): { T }
	local pool = table.clone(items)
	local picked = {}
	while #picked < count and #pool > 0 do
		local choice = WeightedRandom.Pick(pool, weightOf, rng)
		if choice == nil then
			break
		end
		table.insert(picked, choice)
		local index = table.find(pool, choice)
		if index then
			table.remove(pool, index)
		end
	end
	return picked
end

return WeightedRandom
```

```lua
--!strict
--[[
	Janitor
	Path: ReplicatedStorage.Shared.Classes.Janitor

	Tracks everything a system creates (connections, instances, threads, callbacks and
	objects with a Destroy/Disconnect method) and releases it in reverse order of
	registration. Every match, enemy, crop, defense, controller and character session
	owns exactly one Janitor, so tearing an owner down can never leak a connection.

	Dependencies: none.

	API:
		Janitor.new() -> Janitor
		Janitor:Add(task, methodName?) -> task       -- methodName overrides the cleanup method
		Janitor:AddKeyed(key, task, methodName?) -> task  -- replaces (and cleans) a previous task with the same key
		Janitor:Remove(key)                          -- cleans one keyed task now
		Janitor:Detach(key) -> task?                 -- stops tracking a keyed task WITHOUT cleaning it
		Janitor:Connect(signal, handler) -> connection
		Janitor:Cleanup()                            -- cleans everything, Janitor stays usable
		Janitor:Destroy()                            -- cleans everything, later Adds are cleaned immediately
		Janitor:IsDestroyed() -> boolean
]]

type TaskRecord = {
	Task: any,
	Method: string?,
}

local Janitor = {}
Janitor.__index = Janitor

export type Janitor = typeof(setmetatable(
	{} :: {
		_tasks: { TaskRecord },
		_keyed: { [any]: TaskRecord },
		_destroyed: boolean,
		_cleaning: boolean,
	},
	Janitor
))

local function cleanTask(record: TaskRecord)
	local item = record.Task
	local method = record.Method
	local kind = typeof(item)

	if method ~= nil then
		local fn = (item :: any)[method]
		if type(fn) == "function" then
			fn(item)
		end
		return
	end

	if kind == "RBXScriptConnection" then
		(item :: RBXScriptConnection):Disconnect()
	elseif kind == "Instance" then
		(item :: Instance):Destroy()
	elseif kind == "function" then
		(item :: () -> ())()
	elseif kind == "thread" then
		local thread = item :: thread
		if coroutine.status(thread) ~= "dead" and thread ~= coroutine.running() then
			pcall(task.cancel, thread)
		end
	elseif kind == "table" then
		local tbl = item :: any
		if type(tbl.Destroy) == "function" then
			tbl:Destroy()
		elseif type(tbl.Disconnect) == "function" then
			tbl:Disconnect()
		end
	end
end

local function safeClean(record: TaskRecord)
	local ok, err = pcall(cleanTask, record)
	if not ok then
		warn(`[Janitor] cleanup task failed: {err}`)
	end
end

function Janitor.new(): Janitor
	return setmetatable({
		_tasks = {},
		_keyed = {},
		_destroyed = false,
		_cleaning = false,
	}, Janitor)
end

function Janitor.Add<T>(self: Janitor, item: T, methodName: string?): T
	local record: TaskRecord = { Task = item, Method = methodName }
	if self._destroyed then
		-- Anything registered after destruction is released immediately so late async
		-- work (e.g. a model created after its owner died) can never leak.
		safeClean(record)
		return item
	end
	table.insert(self._tasks, record)
	return item
end

function Janitor.AddKeyed<T>(self: Janitor, key: any, item: T, methodName: string?): T
	self:Remove(key)
	local record: TaskRecord = { Task = item, Method = methodName }
	if self._destroyed then
		safeClean(record)
		return item
	end
	self._keyed[key] = record
	return item
end

function Janitor.Remove(self: Janitor, key: any)
	local record = self._keyed[key]
	if record then
		self._keyed[key] = nil
		safeClean(record)
	end
end

-- Hands ownership of a keyed task back to the caller (e.g. a corpse that must
-- outlive its enemy object for a moment).
function Janitor.Detach(self: Janitor, key: any): any
	local record = self._keyed[key]
	if record then
		self._keyed[key] = nil
		return record.Task
	end
	return nil
end

function Janitor.Connect(self: Janitor, signal: any, handler: (...any) -> ()): any
	local connection = signal:Connect(handler)
	self:Add(connection)
	return connection
end

function Janitor.Cleanup(self: Janitor)
	if self._cleaning then
		return
	end
	self._cleaning = true

	for key, record in self._keyed do
		self._keyed[key] = nil
		safeClean(record)
	end

	-- Tasks added while cleaning are appended and picked up by the loop.
	local tasks = self._tasks
	while #tasks > 0 do
		local record = table.remove(tasks) :: TaskRecord
		safeClean(record)
	end

	self._cleaning = false
end

function Janitor.Destroy(self: Janitor)
	self._destroyed = true
	self:Cleanup()
end

function Janitor.IsDestroyed(self: Janitor): boolean
	return self._destroyed
end

return Janitor
```

```lua
		return

	local snapshot = table.clone(self._connections)
```

```lua
	},
	StateMachine
))

function StateMachine.new(definition: Definition): StateMachine
	assert(definition.States[definition.Initial], `[StateMachine:{definition.Name}] initial state is not defined`)
	for from, targets in definition.Transitions do
		assert(from == "*" or definition.States[from], `[StateMachine:{definition.Name}] unknown state {from}`)
		for _, target in targets do
			assert(definition.States[target], `[StateMachine:{definition.Name}] unknown target {target}`)
		end
	end
	return setmetatable({
		Name = definition.Name,
		StateChanged = Signal.new(),
		_definition = definition,
		_state = nil,
		_enteredAt = os.clock(),
		_transitioning = false,
		_pending = {},
		_destroyed = false,
	}, StateMachine)
end

function StateMachine.Start(self: StateMachine, data: any?)
	assert(self._state == nil, `[StateMachine:{self.Name}] already started`)
	self:_apply(self._definition.Initial, data)
end

function StateMachine.CanTransition(self: StateMachine, toState: string): boolean
	if self._destroyed or self._definition.States[toState] == nil then
		return false
	end
	local current = self._state
	if current == nil then
		return false
	end
	local allowed = self._definition.Transitions[current]
	if allowed and table.find(allowed, toState) then
		return true
	end
	local wildcard = self._definition.Transitions["*"]
	return wildcard ~= nil and table.find(wildcard, toState) ~= nil
end

function StateMachine.Transition(self: StateMachine, toState: string, data: any?): boolean
	if self._destroyed then
		return false
	end
	if self._transitioning then
		table.insert(self._pending, { To = toState, Data = data })
		return true
	end
	if not self:CanTransition(toState) then
		warn(`[StateMachine:{self.Name}] rejected transition {tostring(self._state)} -> {toState}`)
		return false
	end
	self:_apply(toState, data)
	return true
end

function StateMachine._apply(self: StateMachine, toState: string, data: any?)
	self._transitioning = true
	local fromState = self._state
	local states = self._definition.States

	if fromState then
		local exit = states[fromState].Exit
		if exit then
			local ok, err = xpcall(exit, debug.traceback, self, toState)
			if not ok then
				warn(`[StateMachine:{self.Name}] Exit({fromState}) failed: {err}`)
			end
		end
	end

	self._state = toState
	self._enteredAt = os.clock()

	local enter = states[toState].Enter
	if enter then
		local ok, err = xpcall(enter, debug.traceback, self, fromState, data)
		if not ok then
			warn(`[StateMachine:{self.Name}] Enter({toState}) failed: {err}`)
		end
	end

	self._transitioning = false
	self.StateChanged:Fire(toState, fromState)

	-- Apply transitions queued by handlers (validated against the new state).
	if #self._pending > 0 and not self._destroyed then
		local nextRequest = table.remove(self._pending, 1) :: { To: string, Data: any? }
		self:Transition(nextRequest.To, nextRequest.Data)
	end
end

function StateMachine.Update(self: StateMachine, dt: number)
	local current = self._state
	if self._destroyed or current == nil or self._transitioning then
		return
	end
	local update = self._definition.States[current].Update
	if update then
		update(self, dt)
	end
end

function StateMachine.GetState(self: StateMachine): string?
	return self._state
end

function StateMachine.Is(self: StateMachine, state: string): boolean
	return self._state == state
end

function StateMachine.TimeInState(self: StateMachine): number
	return os.clock() - self._enteredAt
end

function StateMachine.Destroy(self: StateMachine)
	self._destroyed = true
	table.clear(self._pending)
	self.StateChanged:Destroy()
end

return StateMachine
```

```lua
--!strict
--[[
	RemoteService
	Path: ServerScriptService.Server.Services.RemoteService

	Owns every RemoteEvent / RemoteFunction. Instances are created from NetworkConfig
	under ReplicatedStorage.Remotes during Init, so the network surface is declared in
	exactly one place. Every client->server call passes through one guarded pipeline:

		1. sender must still be in the game
		2. rate limit (token bucket per player per remote)
		3. argument count + schema validation (Shared.Utilities.Validate)
		4. handler runs inside pcall; errors become ServerError, never a crash
		5. violations (spam / malformed) are counted per player

	Handlers receive validated arguments only. They express INTENT; the owning service
	decides outcomes.

	Dependencies: NetworkConfig, ResultCode, Result, Validate, RemoteRateLimiter

	API:
		RemoteService:OnEvent(name, handler(player, ...))
		RemoteService:OnInvoke(name, handler(player, ...) -> Result)
		RemoteService:FireClient(name, player, ...)
		RemoteService:FireClients(name, players, ...)
		RemoteService:FireAllClients(name, ...)
		RemoteService:ReportViolation(player, remoteName, reason)
		RemoteService:GetViolationCount(player) -> number
		RemoteService:VerifyHandlers()      -- Bootstrap, after all services started
]]

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local Shared = ReplicatedStorage.Shared
local NetworkConfig = require(Shared.Config.NetworkConfig)
local ResultCode = require(Shared.Constants.ResultCode)
local Result = require(Shared.Utilities.Result)
local Validate = require(Shared.Utilities.Validate)

local Server = script:FindFirstAncestor("Server") :: Instance
local RemoteRateLimiter = require(Server.Systems.RemoteRateLimiter)

local RemoteService = {}

local remotes: { [string]: Instance } = {}
local handlersBound: { [string]: boolean } = {}
local rateLimiter: any = nil
local violations: { [Player]: { number } } = {}

local function now(): number
	return os.clock()
end

function RemoteService:Init()
	local folder = ReplicatedStorage:FindFirstChild(NetworkConfig.FolderName)
	if folder then
		-- A stale folder (e.g. left in the place file) would shadow ours.
		folder:Destroy()
	end
	folder = Instance.new("Folder")
	folder.Name = NetworkConfig.FolderName

	local limits = {}
	for name, definition in NetworkConfig.Remotes do
		local remote = Instance.new(definition.Class)
		remote.Name = name
		remote.Parent = folder
		remotes[name] = remote
		if definition.RateLimit then
			limits[name] = definition.RateLimit
		end
	end
	folder.Parent = ReplicatedStorage

	rateLimiter = RemoteRateLimiter.new(limits)
end

function RemoteService:Start()
	Players.PlayerRemoving:Connect(function(player)
		rateLimiter:RemovePlayer(player)
		violations[player] = nil
	end)
end

-- Called by Bootstrap after every service has started. A client->server remote with
-- no handler would leave RemoteFunction calls hanging, so it is answered with a
-- failure and reported as a configuration error.
function RemoteService:VerifyHandlers()
	for name, definition in NetworkConfig.Remotes do
		if definition.Direction == "ClientToServer" and not handlersBound[name] then
			warn(`[RemoteService] no handler bound for {name}`)
			local remote = remotes[name]
			if remote:IsA("RemoteFunction") then
				(remote :: RemoteFunction).OnServerInvoke = function()
					return Result.Fail(ResultCode.ServerError)
				end
			end
		end
	end
end

function RemoteService:ReportViolation(player: Player, remoteName: string, reason: string)
	local list = violations[player]
	if list == nil then
		list = {}
		violations[player] = list
	end
	local t = now()
	table.insert(list, t)
	local window = NetworkConfig.Security.ViolationWindow
	while #list > 0 and t - list[1] > window do
		table.remove(list, 1)
	end
	if #list == NetworkConfig.Security.ViolationThreshold then
		warn(`[Security] {player.Name} ({player.UserId}) exceeded violation threshold; last: {remoteName} {reason}`)
		if NetworkConfig.Security.KickOnAbuse then
			player:Kick("Too many invalid requests.")
		end
	end
end

function RemoteService:GetViolationCount(player: Player): number
	local list = violations[player]
	return if list then #list else 0
end

-- Returns (ok, failureCode) for the shared guard pipeline.
local function guard(player: Player, name: string, argCount: number, args: { any }): (boolean, string?)
	if player.Parent ~= Players then
		return false, ResultCode.NotAllowed
	end
	if not rateLimiter:Check(player, name, now()) then
		RemoteService:ReportViolation(player, name, "rate limit")
		return false, ResultCode.RateLimited
	end
	local definition = NetworkConfig.Remotes[name]
	local ok, reason = Validate.CheckArgs(args, argCount, definition.Args or {})
	if not ok then
		RemoteService:ReportViolation(player, name, reason or "invalid")
		return false, ResultCode.InvalidPayload
	end
	return true, nil
end

local function getRemote(name: string, className: string): any
	local remote = remotes[name]
	assert(remote, `[RemoteService] unknown remote {name}`)
	assert(remote.ClassName == className, `[RemoteService] {name} is a {remote.ClassName}, not {className}`)
	return remote
end

function RemoteService:OnEvent(name: string, handler: (Player, ...any) -> ())
	local definition = NetworkConfig.Remotes[name]
	assert(definition and definition.Direction == "ClientToServer", `[RemoteService] {name} is not client->server`)
	assert(not handlersBound[name], `[RemoteService] handler for {name} bound twice`)
	handlersBound[name] = true

	local remote = getRemote(name, "RemoteEvent") :: RemoteEvent
	remote.OnServerEvent:Connect(function(player: Player, ...)
		local argCount = select("#", ...)
		local args = { ... }
		local ok = guard(player, name, argCount, args)
		if not ok then
			return
		end
		local success, err = xpcall(handler, debug.traceback, player, table.unpack(args, 1, argCount))
		if not success then
			warn(`[RemoteService] {name} handler error for {player.Name}: {err}`)
		end
	end)
end

function RemoteService:OnInvoke(name: string, handler: (Player, ...any) -> any)
	local definition = NetworkConfig.Remotes[name]
	assert(definition and definition.Direction == "ClientToServer", `[RemoteService] {name} is not client->server`)
	assert(not handlersBound[name], `[RemoteService] handler for {name} bound twice`)
	handlersBound[name] = true

	local remote = getRemote(name, "RemoteFunction") :: RemoteFunction
	remote.OnServerInvoke = function(player: Player, ...)
		local argCount = select("#", ...)
		local args = { ... }
		local ok, code = guard(player, name, argCount, args)
		if not ok then
			return Result.Fail(code :: string)
		end
		local success, response = xpcall(handler, debug.traceback, player, table.unpack(args, 1, argCount))
		if not success then
			warn(`[RemoteService] {name} handler error for {player.Name}: {response}`)
			return Result.Fail(ResultCode.ServerError)
		end
		if type(response) ~= "table" or type(response.Ok) ~= "boolean" then
			warn(`[RemoteService] {name} handler returned a non-Result value`)
			return Result.Fail(ResultCode.ServerError)
		end
		return response
	end
end

function RemoteService:FireClient(name: string, player: Player, ...: any)
	if player.Parent ~= Players then
		return
	end
	local remote = getRemote(name, "RemoteEvent") :: RemoteEvent
	remote:FireClient(player, ...)
end

function RemoteService:FireClients(name: string, players: { Player }, ...: any)
	local remote = getRemote(name, "RemoteEvent") :: RemoteEvent
	for _, player in players do
		if player.Parent == Players then
			remote:FireClient(player, ...)
		end
	end
end

function RemoteService:FireAllClients(name: string, ...: any)
	local remote = getRemote(name, "RemoteEvent") :: RemoteEvent
	remote:FireAllClients(...)
end

return RemoteService
```

```lua
	RemoteRateLimiter
	Path: ServerScriptService.Server.Systems.RemoteRateLimiter

	Per-player, per-remote token buckets. RemoteService consults it before decoding a
	payload, so spam is rejected at the cheapest possible point.

	Dependencies: Shared.Classes.TokenBucket

	API:
		RemoteRateLimiter.new(limits: { [remoteName]: { Capacity, Refill } }) -> limiter
		limiter:Check(player, remoteName, now) -> boolean
		limiter:RemovePlayer(player)

function RemoteRateLimiter.new(limits: { [string]: Limit })
	return setmetatable({
		_limits = limits,
		_buckets = {} :: { [Player]: { [string]: TokenBucket.TokenBucket } },
	}, RemoteRateLimiter)
end

function RemoteRateLimiter:Check(player: Player, remoteName: string, now: number): boolean
	local perPlayer = self._buckets[player]
	if perPlayer == nil then
		perPlayer = {}
		self._buckets[player] = perPlayer
	end
	local bucket = perPlayer[remoteName]
	if bucket == nil then
		local limit = self._limits[remoteName] or DEFAULT_LIMIT
		bucket = TokenBucket.new(limit.Capacity, limit.Refill, now)
		perPlayer[remoteName] = bucket
	end
	return bucket:TryConsume(now)
end

function RemoteRateLimiter:RemovePlayer(player: Player)
	self._buckets[player] = nil
end

return RemoteRateLimiter
```

```lua
		Migrations.CurrentVersion
		Migrations.Run(data) -> (ok: boolean, reason: string?)   -- mutates data in place

local Migrations = {}

Migrations.CurrentVersion = 1

local Steps: { [number]: (data: { [string]: any }) -> () } = {
	-- 0 -> 1: early builds stored UnlockedCostumes as an array of ids and kept
	-- Candies under "Candy". Convert both to the current shape.
	[0] = function(data)
		local costumes = data.UnlockedCostumes
		if type(costumes) == "table" and #costumes > 0 then
			local set = {}
			for _, costumeId in costumes do
				if type(costumeId) == "string" then
					set[costumeId] = true
				end
			end
			data.UnlockedCostumes = set
		end
		if data.Candies == nil and type(data.Candy) == "number" then
			data.Candies = data.Candy
		end
		data.Candy = nil
	end,
}

function Migrations.Run(data: { [string]: any }): (boolean, string?)
	local version = data.Version
	if version == nil then
		version = 0
	end
	if type(version) ~= "number" or version % 1 ~= 0 or version < 0 then
		return false, `invalid profile version {tostring(version)}`
	end
	if version > Migrations.CurrentVersion then
		return false, `profile version {version} is newer than server version {Migrations.CurrentVersion}`
	end

	while version < Migrations.CurrentVersion do
		local step = Steps[version]
		if step == nil then
			return false, `missing migration step from version {version}`
		end
		step(data)
		version += 1
		data.Version = version
	end
	return true, nil
end
```

```lua
	SessionLockedStore
	Path: ServerScriptService.Server.Data.SessionLockedStore

	Built-in, ProfileService-compatible profile store used when the ProfileService
	package is not installed. It implements the guarantees the game relies on:

	  * Session locking: a profile can be active on only one server. The lock lives in
	    the stored record (MetaData.ActiveSession) and is written atomically with
	    UpdateAsync. A joining server waits for the previous server to release; a lock
	    that has not been refreshed for StaleLockSeconds (crashed server) is taken over.
	  * Lock ownership is re-checked on every save. If another server stole the session,
	    the save is cancelled (never overwriting newer data) and the profile is released
	    with reason "SessionStolen" so the game can kick the player safely.
	  * Saves are serialized per profile: a save requested while one is running is
	    coalesced into a single follow-up save. Released profiles never save again.
	  * Autosave refreshes the lock timestamp of every active profile.

	The backing store only needs UpdateAsync(key, transform), so the in-memory
	MemoryDataStore below can stand in for a DataStore in Studio and in unit tests.

	Dependencies: Shared.Utilities.TableUtil

	-- caller must never be able to mutate stored data by reference.

	self._records[key] = TableUtil.DeepCopy(result)

-- One UpdateAsync round trip. Returns "Saved", "Stolen" or "Failed".
function Profile._writeOnce(self: Profile, release: boolean): string
	local store = self._store
	local stolen = false
	local snapshot = TableUtil.DeepCopy(self.Data)
	local ok, err = pcall(function()
		store._dataStore:UpdateAsync(self.Key, function(old: Record?)
			local record: Record = old
				or {
					Data = {},
					MetaData = {
						ActiveSession = nil,
						LastUpdate = 0,
						SessionLoadCount = 0,
						FormatVersion = FORMAT_VERSION,
					},
				}
			local session = record.MetaData.ActiveSession
			if session ~= nil and session.SessionId ~= store._sessionId then
				-- Another server owns this profile now. Cancel the write.
				stolen = true
				return nil
			end
			record.Data = snapshot
			record.MetaData.LastUpdate = store._clock()
			record.MetaData.ActiveSession = if release
				then nil
				else { SessionId = store._sessionId, PlaceId = game.PlaceId }
			return record, self._userIds
		end)
	end)
	if not ok then
		warn(`[SessionLockedStore] save of {self.Key} failed: {err}`)
		return "Failed"
	end
	return if stolen then "Stolen" else "Saved"
end

function Profile._runSave(self: Profile, release: boolean): boolean
	local store = self._store
	if self.IsTemporary then
		if release then
			self:_markReleased("Released")
		end
		return true
	end

	local outcome = "Failed"
	for attempt = 1, store._saveAttempts do
		outcome = self:_writeOnce(release)
		if outcome ~= "Failed" then
			break
		end
		if attempt < store._saveAttempts then
			store._wait(2 ^ attempt)
		end
	end

	if outcome == "Stolen" then
		warn(`[SessionLockedStore] session for {self.Key} was taken by another server; releasing locally`)
		self:_markReleased("SessionStolen")
		return false
	end
	if release then
		-- Even if the final write failed the profile must stop being used here; the
		-- stale-lock timeout lets the next server take over safely.
		self:_markReleased(if outcome == "Saved" then "Released" else "ReleasedWithError")
	end
	return outcome == "Saved"
end

function Profile.SaveAsync(self: Profile): boolean
	if not self._active or self._releasing then
		return false
	end
	if self._saving then
		self._saveQueued = true
		return true
	end
	self._saving = true
	local ok = self:_runSave(false)
	-- Coalesce saves requested while this one was in flight into one more write.
	while self._saveQueued and self._active and not self._releasing do
		self._saveQueued = false
		ok = self:_runSave(false)
	end
	self._saving = false
	return ok
end

function Profile.Save(self: Profile)
	task.spawn(function()
		self:SaveAsync()
	end)
end

function Profile.ReleaseAsync(self: Profile)
	if not self._active then
		return
	end
	if self._releasing then
		-- Another thread is already releasing; wait for it.
		local event = self._releasedEvent
		if event then
			event.Event:Wait()
		end
		return
	end
	self._releasing = true
	local event = Instance.new("BindableEvent")
	self._releasedEvent = event
	-- Let an in-flight save finish first so writes stay ordered.
	while self._saving do
		self._store._wait(0.1)
	end
	self:_runSave(true)
	event:Fire()
	event:Destroy()
	self._releasedEvent = nil
end

function Profile.Release(self: Profile)
	task.spawn(function()
		self:ReleaseAsync()
	end)
end

-- Attempts to acquire the session lock once. Returns ("Acquired", record) |
-- ("Locked", nil) | ("Error", message).
function SessionLockedStore:_tryAcquire(key: string, userId: number): (string, any)
	local now = self._clock()
	local sessionId = self._sessionId
	local staleAfter = self._staleLock
	local status = "Error"
	local acquiredRecord: Record? = nil

	local ok, err = pcall(function()
		self._dataStore:UpdateAsync(key, function(old: Record?)
			local record: Record = old
				or {
					Data = TableUtil.DeepCopy(self._template),
					MetaData = {
						ActiveSession = nil,
						LastUpdate = 0,
						SessionLoadCount = 0,
						FormatVersion = FORMAT_VERSION,
					},
				}
			-- Older raw records without MetaData are wrapped instead of discarded.
			if type(record.MetaData) ~= "table" then
				record = {
					Data = record :: any,
					MetaData = { ActiveSession = nil, LastUpdate = 0, SessionLoadCount = 0, FormatVersion = FORMAT_VERSION },
				}
			end
			local session = record.MetaData.ActiveSession
			local lockIsFresh = (now - (record.MetaData.LastUpdate or 0)) < staleAfter
			if session ~= nil and session.SessionId ~= sessionId and lockIsFresh then
				status = "Locked"
				return nil
			end
			record.MetaData.ActiveSession = { SessionId = sessionId, PlaceId = game.PlaceId }
			record.MetaData.LastUpdate = now
			record.MetaData.SessionLoadCount = (record.MetaData.SessionLoadCount or 0) + 1
			status = "Acquired"
			acquiredRecord = record
			return record, { userId }
		end)
	end)

	if not ok then
		return "Error", tostring(err)
	end
	return status, acquiredRecord
end

function SessionLockedStore:LoadProfileAsync(key: string, userId: number): (Profile?, string?)
	if self._profiles[key] or self._loading[key] then
		return nil, "profile already loaded or loading on this server"
	end
	self._loading[key] = true

	local record: Record? = nil
	local lastProblem = "unknown"

	if self._isTemporary then
		record = {
			Data = TableUtil.DeepCopy(self._template),
			MetaData = { ActiveSession = nil, LastUpdate = 0, SessionLoadCount = 1, FormatVersion = FORMAT_VERSION },
		}
	else
		for attempt = 1, self._loadAttempts do
			local status, result = self:_tryAcquire(key, userId)
			if status == "Acquired" then
				record = result
				break
			elseif status == "Locked" then
				lastProblem = "profile is active on another server"
			else
				lastProblem = result
				warn(`[SessionLockedStore] load of {key} attempt {attempt} failed: {result}`)
			end
			if attempt < self._loadAttempts then
				self._wait(self._loadRetryDelay)
			end
		end
	end

	self._loading[key] = nil
	if record == nil then
		return nil, lastProblem
	end

	local profile = setmetatable({
		Key = key,
		UserId = userId,
		Data = record.Data,
		IsTemporary = self._isTemporary,
		_store = self,
		_active = true,
		_releasing = false,
		_saving = false,
		_saveQueued = false,
		_releaseListeners = {},
		_userIds = { userId },
		_releasedEvent = nil,
	}, Profile) :: Profile
	self._profiles[key] = profile
	return profile, nil
end

function SessionLockedStore:SaveAllAsync()
	for _, profile in self:GetActiveProfiles() do
		task.spawn(function()
			profile:SaveAsync()
		end)
	end
end

function SessionLockedStore:ReleaseAllAsync(timeoutSeconds: number)
	local profiles = self:GetActiveProfiles()
	local remaining = #profiles
	if remaining == 0 then
		return
	end
	local done = Instance.new("BindableEvent")
	for _, profile in profiles do
		task.spawn(function()
			profile:ReleaseAsync()
			remaining -= 1
			if remaining == 0 then
				done:Fire()
			end
		end)
	end
	local timeoutThread = task.delay(timeoutSeconds, function()
		done:Fire()
	end)
	if remaining > 0 then
		done.Event:Wait()
	end
	pcall(task.cancel, timeoutThread)
	done:Destroy()
end

return SessionLockedStore
```

```lua
function DataService:LoadProfileAsync(player: Player): (boolean, string?)
	if profiles[player] then
		return true, nil
	end
	player:SetAttribute(Attributes.ProfileStatus, "Loading")

	local profile, problem = store:LoadProfileAsync(player.UserId)
	if profile == nil then
		player:SetAttribute(Attributes.ProfileStatus, "Failed")
		return false, problem or "unknown error"
	end

	if player.Parent ~= Players then
		-- Player left while the profile was loading: release immediately.
		profile:Release()
		return false, "player left during load"
	end

	local migrated, reason = Migrations.Run(profile.Data)
	if not migrated then
		-- Refuse to use (and therefore to overwrite) data we cannot understand.
		warn(`[DataService] migration failed for {player.UserId}: {reason}`)
		profile:Release()
		player:SetAttribute(Attributes.ProfileStatus, "Failed")
		return false, reason
	end

	profile:AddUserId(player.UserId)
	profile:Reconcile()

	-- Every class flagged DefaultUnlocked is owned, even if added after the profile
	-- was created (reconcile alone does not add keys into existing sets).
	for classId, definition in ClassConfig.Classes do
		if definition.DefaultUnlocked then
			profile.Data.UnlockedCostumes[classId] = true
		end
	end

	profile:ListenToRelease(function(_releaseReason: string)
		-- ReleaseProfile clears the entry before releasing, so if it is still ours the
		-- release came from elsewhere (session taken by another server, or a backend
		-- shutdown). The player must not keep playing without a writable profile.
		if profiles[player] == profile then
			profiles[player] = nil
			if player.Parent == Players then
				player:Kick("Your data was loaded on another server. Please rejoin.")
			end
		end
	end)

	profiles[player] = profile
	player:SetAttribute(Attributes.ProfileStatus, if profile.IsTemporary then "Temporary" else "Loaded")

-- snapshot so a half-applied change can never be saved.

	local data = self:GetData(player)

		return false, nil
	end

		table.clear(data)
		for key, value in snapshot do

		return false, nil
	end
	markChanged(player)
	return true, result
end
```

```lua
-- Applies class armour and status effects (spawn protection) before damage.
function PlayerSession:TakeDamage(amount: number, _source: any): number
	local humanoid = self.Humanoid
	if not self:IsAlive() or humanoid == nil or amount <= 0 then
		return 0
	end
	local classDefinition = self:GetClassDefinition()
	local multiplier = classDefinition.BaseStats.DamageTakenMultiplier
		* Services.StatusEffectService:GetDamageTakenMultiplier(self)
	local dealt = math.min(humanoid.Health, amount * multiplier)
	if dealt <= 0 then
		return 0
	end
	humanoid.Health -= dealt
	return dealt
end

function PlayerSession:Heal(amount: number): number
	local humanoid = self.Humanoid
	if not self:IsAlive() or humanoid == nil or amount <= 0 then
		return 0
	end
	local healed = math.min(amount, humanoid.MaxHealth - humanoid.Health)
	humanoid.Health += healed
	return healed
end

local function accessoryStat(self: any, statName: string): number
	local total = 0
	for itemId in self.Accessories do
		local item = DefenseConfig.Items[itemId]
		if item and item.Stats and item.Stats[statName] then
			total += item.Stats[statName]
		end
	end
	return total
end

end

-- Recomputes humanoid stats from class base stats + accessories + status effects.
-- Called on spawn, accessory purchase and whenever status effects change.
function PlayerSession:RecomputeStats()
	local humanoid = self.Humanoid
	if humanoid == nil or humanoid.Parent == nil then
		return
	end
	local base = self:GetClassDefinition().BaseStats
	local effects = Services.StatusEffectService

	local inMatch = self.Match ~= nil
	local maxHealth = base.MaxHealth + (if inMatch then accessoryStat(self, "MaxHealth") else 0)
	local walkSpeed = (base.WalkSpeed + (if inMatch then accessoryStat(self, "WalkSpeed") else 0))
		* effects:GetSpeedMultiplier(self)
	local jumpHeight = if effects:BlocksJump(self) then 0 else base.JumpHeight

	if humanoid.Health > 0 then
		local previousMax = humanoid.MaxHealth
		humanoid.MaxHealth = maxHealth
		if maxHealth > previousMax then
			humanoid.Health = math.min(maxHealth, humanoid.Health + (maxHealth - previousMax))
		else
			humanoid.Health = math.min(humanoid.Health, maxHealth)
		end
	end
	humanoid.WalkSpeed = walkSpeed
	humanoid.UseJumpPower = false
	humanoid.JumpHeight = jumpHeight
end

function PlayerSession:OnStatusEffectsChanged()
	self:RecomputeStats()
end

-- Clears everything that only exists inside a match.
function PlayerSession:ResetMatchState()
	self.Match = nil
	table.clear(self.Accessories)
	table.clear(self.UtilityReadyAt)
	self.LastAttackSequence = -1
	self.MatchStats = PlayerSession._emptyStats()
	self.StatsCommitted = false
	self.Player:SetAttribute(Attributes.MatchId, nil)
end

function PlayerSession:Destroy()
	if self.Destroyed then
		return
	end
	self.Destroyed = true
	self.Janitor:Destroy()
end
```

```lua
-- Loads a character and places it. A SpawnToken guards against overlapping spawn
-- requests (e.g. a lobby respawn racing a match start): only the latest one wins.
function PlayerService:SpawnAsync(session: any, cframe: CFrame): boolean
	local player: Player = session.Player
	session.SpawnToken += 1
	local token = session.SpawnToken

	local ok, err = pcall(function()
		player:LoadCharacter()
	end)
	if not ok then
		warn(`[PlayerService] LoadCharacter failed for {player.Name}: {err}`)
		return false
	end
	if session.Destroyed or session.SpawnToken ~= token or player.Parent ~= Players then
		return false
	end

	local character = player.Character
	if character == nil then
		return false
	end
	local root = character:WaitForChild("HumanoidRootPart", 10)
	if root == nil or session.SpawnToken ~= token then
		return false
	end
	if not session:BindCharacter(character) then
		return false
	end

	-- Players pass through their own defenses (see CollisionGroups); parts added later
	-- (accessories, the weapon) join the group too.
	CollisionGroups.SetModel(character, CollisionGroup.Players)
	session.CharacterJanitor:Connect(character.DescendantAdded, function(descendant)
		if descendant:IsA("BasePart") then
			descendant.CollisionGroup = CollisionGroup.Players
		end
	end)
	character:PivotTo(cframe)
	local humanoid = session.Humanoid :: Humanoid
	session.CharacterJanitor:Connect(humanoid.Died, function()
		if session.Character == character then
			onCharacterDied(session)
		end
	end)
	return true
end

local function onPlayerAdded(player: Player)
	if sessions[player] then
		return
	end
	local session = PlayerSession.new(player)
	sessions[player] = session

	-- (it can take a while if another server still holds the session lock). Queueing

	Services.ClassService:InitializePlayer(session)
	Services.ClassService:ApplyToCharacter(session)
```

```lua
function Match:_enterState(state: string)

function Match:_buildStateMachine()
	local phases = MatchConfig.Phases
	local states = {}

	states[MatchState.Waiting] = {
		Enter = function()
			self:_enterState(MatchState.Waiting)
			self:SetObjective("Get ready...")
			self:SetPhaseDuration(phases.WaitingTimeout)
			local index = 0
			for _, session in self.Sessions do
				index += 1
				self:_spawnSession(session, index)
			end
		end,
		Update = function(machine)
			local everyoneReady = true
			for _, session in self.Sessions do
				if not session:IsAlive() then
					everyoneReady = false
					break
				end
			end
			if everyoneReady or self:_phaseExpired() then
				machine:Transition(MatchState.Starting)
			end
		end,
	}

	states[MatchState.Starting] = {
		Enter = function()
			self:_enterState(MatchState.Starting)
			self:SetObjective("Plant seeds and build defenses before the zombies arrive!")
			self:SetPhaseDuration(phases.Starting)
			for player, _ in self.Sessions do
				local startingCoins = EconomyConfig.MatchCoins.Starting + Services.UpgradeService:GetStartingCoinBonus(player)
				Services.MatchCoinService:Reset(player, startingCoins)
				Services.SeedService:GrantSoftLockIfNeeded(player)
			end
			Services.SeedShopService:Restock(self)
			self:Notify("Protect the Pumpkin Farm! Plant seeds, then survive the night.", "Info", 6)
		end,
		Update = function(machine)
			if self:_phaseExpired() then
				machine:Transition(MatchState.Intermission)
			end
		end,
	}

	states[MatchState.Intermission] = {
		Enter = function()
			self:_enterState(MatchState.Intermission)
			local duration = if self._firstIntermission then phases.FirstIntermission else phases.Intermission
			if not self._firstIntermission then
				Services.SeedShopService:Restock(self)
			end
			self._firstIntermission = false
			self:SetObjective("Intermission: buy seeds, plant, harvest and fortify!")
			self:SetPhaseDuration(duration)
		end,
		Update = function(machine)
			if self:_phaseExpired() then
				machine:Transition(MatchState.PreparingWave)
			end
		end,
	}

	states[MatchState.PreparingWave] = {
		Enter = function()
			self:_enterState(MatchState.PreparingWave)
			self.Wave += 1
			self.IsBossWave = Services.DifficultyService:IsBossWave(self.Wave)
			self:SetAttribute(Attributes.Wave, self.Wave)
			self:SetAttribute(Attributes.IsBossWave, self.IsBossWave)
			Services.WaveService:PrepareWave(self)
			self:SetPhaseDuration(phases.PreparingWave)
			if self.IsBossWave then
				self:SetObjective("THE CURSED HARVESTER APPROACHES!")
				self:FireAll("MatchEvent", "BossIncoming", { Wave = self.Wave })
			else
				self:SetObjective(`Wave {self.Wave} incoming!`)
				self:FireAll("MatchEvent", "WaveIncoming", { Wave = self.Wave })
			end
		end,
		Update = function(machine)
			if self:_phaseExpired() then
				machine:Transition(if self.IsBossWave then MatchState.BossWave else MatchState.WaveActive)
			end
		end,
	}

	local function combatState(stateName: string)
		return {
			Enter = function()
				self:_enterState(stateName)
				self:SetPhaseDuration(nil)
				self:SetObjective(if stateName == MatchState.BossWave
					then "Defeat the Cursed Harvester and protect the crops!"
					else "Defend the farm! Harvest ripe crops when you can.")
				Services.WaveService:StartWave(self)
				self:FireAll("MatchEvent", "WaveStarted", { Wave = self.Wave, IsBoss = stateName == MatchState.BossWave })
			end,
			Update = function(machine, dt)
				Services.WaveService:Update(self, dt)
				if Services.WaveService:IsWaveCleared(self) then
					machine:Transition(MatchState.WaveComplete)
				end
			end,
		}
	end
	states[MatchState.WaveActive] = combatState(MatchState.WaveActive)
	states[MatchState.BossWave] = combatState(MatchState.BossWave)

	states[MatchState.WaveComplete] = {
		Enter = function()
			self:_enterState(MatchState.WaveComplete)
			self:SetPhaseDuration(phases.WaveComplete)
			self:SetObjective(`Wave {self.Wave} survived!`)
			Services.RewardService:OnWaveCompleted(self)
			self:FireAll("MatchEvent", "WaveCompleted", { Wave = self.Wave })
			self:_reviveDowned()
		end,
		Update = function(machine)
			if self:_phaseExpired() then
				if self.Wave % MatchConfig.ExtractionEveryWaves == 0 then
					machine:Transition(MatchState.Extraction)
				else
					machine:Transition(MatchState.Intermission)
				end
			end
		end,
	}

	states[MatchState.Extraction] = {
		Enter = function()
			self:_enterState(MatchState.Extraction)
			self:SetPhaseDuration(phases.ExtractionDecision)
			self:SetObjective("The Escape Door is open! ESCAPE with the harvest, or CONTINUE for more.")
			Services.EscapeService:OpenDoor(self)
		end,
		Update = function(machine)
			local resolution = Services.EscapeService:Update(self, self:_phaseExpired())
			if resolution == "Continue" then
				machine:Transition(MatchState.ContinuePreparation)
			elseif resolution == "End" then
				self.EndReason = "Extracted"
				machine:Transition(MatchState.MatchEnding, "Extracted")
			end
		end,
		Exit = function()
			Services.EscapeService:CloseDoor(self)
		end,
	}

	states[MatchState.ContinuePreparation] = {
		Enter = function()
			self:_enterState(MatchState.ContinuePreparation)
			local difficulty = Services.DifficultyService
			self.ContinueLevel = math.min(self.ContinueLevel + 1, DifficultyConfig.Continue.MaxLevel)
			local danger = difficulty:GetDangerMultiplier(self.ContinueLevel)
			self:SetAttribute(Attributes.ContinueLevel, self.ContinueLevel)
			self:SetAttribute(Attributes.DifficultyMultiplier, danger)
			self:SetPhaseDuration(phases.ContinuePreparation)
			self:SetObjective(`DANGER LEVEL {Format.Roman(self.ContinueLevel)}: prepare! Zombies are {Format.Multiplier(danger)} tougher.`)

			local recovery = MatchConfig.ContinueRecovery
			if recovery.HealPlayers then
				for _, session in self.Sessions do
					local humanoid = session.Humanoid
					if humanoid and session:IsAlive() then
						humanoid.Health = humanoid.MaxHealth
					end
				end
			end
			Services.DefenseService:RepairAll(self, recovery.RepairDefensesFraction)
			self.Core:Heal(recovery.CoreHealFraction)
			Services.CropService:HealAll(self)
			Services.SeedShopService:Restock(self)
			self:FireAll("MatchEvent", "ContinueStarted", { ContinueLevel = self.ContinueLevel, Danger = danger })
		end,
		Update = function(machine)
			if self:_phaseExpired() then
				machine:Transition(MatchState.PreparingWave)
			end
		end,
	}

	states[MatchState.MatchEnding] = {
		Enter = function(_machine, _from, reason)
			local endReason = reason or self.EndReason or "Abandoned"
			self.EndReason = endReason
			self:_enterState(MatchState.MatchEnding)
			self:SetPhaseDuration(phases.MatchEnding)
			self:SetObjective(END_MESSAGES[endReason] or "The match is over.")
			Services.WaveService:StopWave(self)
			if LOSS_REASONS[endReason] then
				Services.HarvestChestService:Forfeit(self)
			end
			for _, session in self.Sessions do
				Services.RewardService:CommitMatchStats(session, self, false)
			end
			self:FireAll("MatchEvent", "MatchEnded", {
				Reason = endReason,
				Wave = self.Wave,
				ContinueLevel = self.ContinueLevel,
				Lost = LOSS_REASONS[endReason] == true,
			})
		end,
		Update = function(machine)
			if self:_phaseExpired() or next(self.Sessions) == nil then
				machine:Transition(MatchState.Cleanup)
			end
		end,
	}

	states[MatchState.Cleanup] = {
		Enter = function()
			self:_enterState(MatchState.Cleanup)
			self:SetPhaseDuration(nil)
			-- Defer so the state machine finishes its transition before teardown.
			task.defer(function()
				Services.MatchService:DestroyMatch(self)
			end)
		end,
	}

	return StateMachine.new({
		Name = `Match_{self.Id}`,
		Initial = MatchState.Waiting,
		States = states,
		Transitions = {
			[MatchState.Waiting] = { MatchState.Starting, MatchState.MatchEnding },
			[MatchState.Starting] = { MatchState.Intermission, MatchState.MatchEnding },
			[MatchState.Intermission] = { MatchState.PreparingWave, MatchState.MatchEnding },
			[MatchState.PreparingWave] = { MatchState.WaveActive, MatchState.BossWave, MatchState.MatchEnding },
			[MatchState.WaveActive] = { MatchState.WaveComplete, MatchState.MatchEnding },
			[MatchState.BossWave] = { MatchState.WaveComplete, MatchState.MatchEnding },
			[MatchState.WaveComplete] = { MatchState.Intermission, MatchState.Extraction, MatchState.MatchEnding },
			[MatchState.Extraction] = { MatchState.ContinuePreparation, MatchState.MatchEnding },
			[MatchState.ContinuePreparation] = { MatchState.PreparingWave, MatchState.MatchEnding },
			[MatchState.MatchEnding] = { MatchState.Cleanup },
			[MatchState.Cleanup] = {},
		},
	})
end

-- Lifecycle --------------------------------------------------------------------------------

function Match:Start()
	self.Machine:Start()
end

function Match:Update(dt: number)
	if self.Destroyed then
		return
	end
	self.Machine:Update(dt)
end

function Match:Destroy()
	if self.Destroyed then
		return
	end
	self.Destroyed = true
	self.Janitor:Destroy()
end

return Match
```

```lua
--[[
	MatchService
	Path: ServerScriptService.Server.Services.MatchService

	Creates, drives and destroys Match instances. Responsibilities:
	  * arena slot allocation (MatchConfig.MaxConcurrentMatches arenas per server)
	  * per-match service initialisation in a fixed order (MATCH_SERVICES) and teardown
	    in reverse order through CleanupService
	  * one central update loop for every match at MatchConfig.UpdateRate
	  * moving players out of a match (escape, disconnect) and back to the lobby

	Multi-server note: CreateMatch(sessions) is the single entry point. A dedicated
	match place would call it from PlayerAdded with the party from TeleportData; nothing
	else in the match code assumes the lobby lives in the same server.

	Dependencies: Match, MatchConfig, LifeState; Registry: PlayerService, CleanupService,
	              RewardService, MatchCoinService, WeaponService, StatusEffectService, CropService,
	              and every service in MATCH_SERVICES

	API:
		MatchService:CanCreateMatch() -> boolean
		MatchService:CreateMatch(sessions) -> Match?
		MatchService:GetMatch(matchId) -> Match?
		MatchService:GetMatches() -> { Match }
		MatchService:GetMatchForPlayer(player) -> Match?
		MatchService:RemoveSession(session, reason)    -- "Escaped" | "Left"
		MatchService:OnPlayerRemoving(session)
		MatchService:DestroyMatch(match)
		MatchService.MatchServices : { string }        -- init order
]]

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")

local Shared = ReplicatedStorage.Shared
local LifeState = require(Shared.Constants.LifeState)
local MatchConfig = require(Shared.Config.MatchConfig)

local Server = script:FindFirstAncestor("Server") :: Instance
local Services = require(Server.Registry)
local Match = require(Server.Classes.Match)

local MatchService = {}

-- Services that hold per-match state. InitMatch runs in this order, CleanupMatch in
-- reverse, so e.g. enemies are removed before the crops and defenses they target.
MatchService.MatchServices = {
	"CropService",
	"SeedShopService",
	"HarvestChestService",
	"BarricadeService",
	"SentryService",
	"DefenseService",
	"EnemyService",
	"WaveService",
	"EscapeService",
}

local matches: { [number]: any } = {}
local arenaSlots: { [number]: number } = {} -- [arenaIndex] = matchId
local nextMatchId = 0

function MatchService:Init() end

function MatchService:Start()
	local step = 1 / MatchConfig.UpdateRate
	local accumulator = 0
	RunService.Heartbeat:Connect(function(dt)
		accumulator += dt
		if accumulator < step then
			return
		end
		local elapsed = accumulator
		accumulator = 0
		for _, match in matches do
			local ok, err = xpcall(match.Update, debug.traceback, match, elapsed)
			if not ok then
				warn(`[MatchService] match {match.Id} update error: {err}`)
			end
		end
	end)
end

function MatchService:_freeArena(): number?
	for index = 1, MatchConfig.MaxConcurrentMatches do
		if arenaSlots[index] == nil then
			return index
		end
	end
	return nil
end

function MatchService:CanCreateMatch(): boolean
	return self:_freeArena() ~= nil
end

function MatchService:CreateMatch(sessions: { any }): any
	local arenaIndex = self:_freeArena()
	if arenaIndex == nil or #sessions == 0 then
		return nil
	end
	nextMatchId += 1
	local matchId = nextMatchId
	arenaSlots[arenaIndex] = matchId

	local origin = MatchConfig.Arena.Origin + Vector3.new(MatchConfig.Arena.Spacing * (arenaIndex - 1), 0, 0)
	local ok, matchOrError = pcall(Match.new, matchId, arenaIndex, origin, sessions)
	if not ok then
		warn(`[MatchService] failed to build match {matchId}: {matchOrError}`)
		arenaSlots[arenaIndex] = nil
		-- Match.new may have attached sessions before failing; detach them again.
		for _, session in sessions do
			session:ResetMatchState()
		end
		return nil
	end
	local match = matchOrError
	matches[matchId] = match

	for _, serviceName in self.MatchServices do
		local service = Services[serviceName]
		local initOk, err = xpcall(service.InitMatch, debug.traceback, service, match)
		if not initOk then
			warn(`[MatchService] {serviceName}:InitMatch failed for match {matchId}: {err}`)
		end
	end

	for _, session in sessions do
		session:SetLifeState(LifeState.InMatch)
	end
	match:Start()
	return match
end

function MatchService:GetMatch(matchId: number): any
	return matches[matchId]
end

function MatchService:GetMatches(): { any }
	local list = {}
	for _, match in matches do
		table.insert(list, match)
	end
	return list
end

function MatchService:GetMatchForPlayer(player: Player): any
	local session = Services.PlayerService:GetSession(player)
	return session and session.Match
end

-- Strips match-only state from a session (coins, weapon, effects, accessories).
local function detach(session: any, match: any)
	Services.CropService:ReleasePlots(match, session.Player)
	match:DetachSession(session)
	Services.MatchCoinService:Clear(session.Player)
	Services.WeaponService:UnequipWeapon(session)
	Services.StatusEffectService:ClearTarget(session)
	session:ResetMatchState()
end

function MatchService:RemoveSession(session: any, reason: string)
	local match = session.Match
	if match == nil then
		return
	end
	Services.RewardService:CommitMatchStats(session, match, reason == "Escaped")
	detach(session, match)
	if reason ~= "Left" and not session.Destroyed then
		session:SetLifeState(LifeState.Lobby)
		task.spawn(function()
			Services.PlayerService:SpawnInLobby(session)
		end)
	end
end

function MatchService:OnPlayerRemoving(session: any)
	if session.Match then
		self:RemoveSession(session, "Left")
	end
end

function MatchService:DestroyMatch(match: any)
	if matches[match.Id] ~= match then
		return
	end
	-- Anyone still inside (match ended normally) goes back to the lobby first.
	for _, session in match:GetSessions() do
		self:RemoveSession(session, "Returned")
	end
	Services.CleanupService:CleanupMatch(match, self.MatchServices)
	matches[match.Id] = nil
	if arenaSlots[match.ArenaIndex] == match.Id then
		arenaSlots[match.ArenaIndex] = nil
	end
end

return MatchService
```

```lua
local function getRemote(name: string): Instance
	local remote = cache[name]
	if remote then
		return remote
	end
	assert(NetworkConfig.Remotes[name], `[Net] unknown remote {name}`)
	local found = folder:WaitForChild(name, 30)

	remote:FireServer(...)
end

	local remote = getRemote(name) :: RemoteEvent
	return remote.OnClientEvent:Connect(handler)
```

```lua
--[[
	ClientState
	Path: StarterPlayerScripts.Client.Modules.ClientState

	Single read-only cache of server-replicated state for all controllers:
	  * Profile / Shop / Chest / Extraction tables from their remotes (+ a snapshot
	    request on start and whenever the player enters a match, so nothing is missed)

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Workspace = game:GetService("Workspace")

local Net = require(script.Parent.Net)

local player = Players.LocalPlayer

type Observer = { Name: string, Callback: (any) -> (), Connection: RBXScriptConnection? }

local ClientState = {
	Profile = nil :: any,
	Shop = nil :: any,
	Chest = nil :: any,
	Extraction = nil :: any,
	MatchFolder = nil :: Folder?,
	ProfileChanged = Signal.new(),
	ShopChanged = Signal.new(),
	ChestChanged = Signal.new(),
	ExtractionChanged = Signal.new(),
	MatchChanged = Signal.new(),
}

local observers: { Observer } = {}
local bindToken = 0

local function connectObserver(observer: Observer)
	if observer.Connection then
		observer.Connection:Disconnect()
		observer.Connection = nil
	end
	local folder = ClientState.MatchFolder
	if folder then
		observer.Connection = folder:GetAttributeChangedSignal(observer.Name):Connect(function()
			observer.Callback(folder:GetAttribute(observer.Name))
		end)
		observer.Callback(folder:GetAttribute(observer.Name))
	else
		observer.Callback(nil)
	end
end

local function setMatchFolder(folder: Folder?)
	if ClientState.MatchFolder == folder then
		return
	end
	ClientState.MatchFolder = folder
	for _, observer in observers do
		connectObserver(observer)
	end
	ClientState.MatchChanged:Fire(folder)
end

local function bindMatch()
	bindToken += 1
	local token = bindToken
	local matchId = player:GetAttribute(Attributes.MatchId)
	if matchId == nil then
		setMatchFolder(nil)
		ClientState.Shop = nil
		ClientState.Chest = nil
		ClientState.Extraction = nil
		ClientState.ShopChanged:Fire(nil)
		ClientState.ChestChanged:Fire(nil)
		ClientState.ExtractionChanged:Fire(nil)
		return
	end
	task.spawn(function()
		local matches = Workspace:WaitForChild("Matches", 30)
		local folder = matches and matches:WaitForChild(`Match_{matchId}`, 30)
		if token ~= bindToken then
			return
		end
		setMatchFolder(folder :: Folder?)
		ClientState.RefreshSnapshot()
	end)
end

function ClientState.RefreshSnapshot()
	task.spawn(function()
		local result = Net:Invoke("RequestSnapshot")
		if not result.Ok or result.Data == nil then
			return
		end
		local data = result.Data
		if data.Profile then
			ClientState.Profile = data.Profile
			ClientState.ProfileChanged:Fire(data.Profile)
		end
		ClientState.Shop = data.Shop
		ClientState.ShopChanged:Fire(data.Shop)
		ClientState.Chest = data.Chest
		ClientState.ChestChanged:Fire(data.Chest)
		ClientState.Extraction = data.Extraction
		ClientState.ExtractionChanged:Fire(data.Extraction)
	end)
end

function ClientState.InMatch(): boolean
	return ClientState.MatchFolder ~= nil
end

function ClientState.GetMatchAttribute(name: string): any
	local folder = ClientState.MatchFolder
	return folder and folder:GetAttribute(name)
end

function ClientState.ObserveMatchAttribute(name: string, callback: (any) -> ()): () -> ()
	local observer: Observer = { Name = name, Callback = callback, Connection = nil }
	table.insert(observers, observer)
	connectObserver(observer)
	return function()
		local index = table.find(observers, observer)
		if index then
			table.remove(observers, index)
		end
		if observer.Connection then
			observer.Connection:Disconnect()
		end
	end
end

function ClientState.Start()
	Net:On("ProfileUpdated", function(profile)
		ClientState.Profile = profile
		ClientState.ProfileChanged:Fire(profile)
	end)
	Net:On("ShopUpdated", function(shop)
		ClientState.Shop = shop
		ClientState.ShopChanged:Fire(shop)
	end)
	Net:On("ChestUpdated", function(chest)
		ClientState.Chest = chest
		ClientState.ChestChanged:Fire(chest)
	end)
	Net:On("ExtractionUpdated", function(extraction)
		ClientState.Extraction = extraction
		ClientState.ExtractionChanged:Fire(extraction)
	end)
	player:GetAttributeChangedSignal(Attributes.MatchId):Connect(bindMatch)
	bindMatch()
	ClientState.RefreshSnapshot()
end

return ClientState
```
