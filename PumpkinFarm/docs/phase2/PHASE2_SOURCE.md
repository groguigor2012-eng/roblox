```lua
--!strict
--[[
	MatchService
	Path: ServerScriptService.Server.Services.MatchService

	Creates, drives and destroys Match instances. Responsibilities:
	  * arena slot allocation (MatchConfig.MaxConcurrentMatches arenas per server)
	  * per-match service initialisation in a fixed order (MATCH_SERVICES) and teardown
	    in reverse order through CleanupService
	  * one central update loop for every match at MatchConfig.UpdateRate
	  * moving players out of a match (escape, disconnect) and back to the lobby

	Multi-server note: CreateMatch(sessions) is the single entry point. A dedicated
	match place would call it from PlayerAdded with the party from TeleportData; nothing
	else in the match code assumes the lobby lives in the same server.

	Dependencies: Match, MatchConfig, LifeState; Registry: PlayerService, CleanupService,
	              RewardService, MatchCoinService, WeaponService, StatusEffectService, CropService,
	              and every service in MATCH_SERVICES

	API:
		MatchService:CanCreateMatch() -> boolean
		MatchService:CreateMatch(sessions) -> Match?
		MatchService:GetMatch(matchId) -> Match?
		MatchService:GetMatches() -> { Match }
		MatchService:GetMatchForPlayer(player) -> Match?
		MatchService:RemoveSession(session, reason)    -- "Escaped" | "Left"
		MatchService:OnPlayerRemoving(session)
		MatchService:DestroyMatch(match)
		MatchService.MatchServices : { string }        -- init order
]]

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")

local Shared = ReplicatedStorage.Shared
local LifeState = require(Shared.Constants.LifeState)
local MatchConfig = require(Shared.Config.MatchConfig)

local Server = script:FindFirstAncestor("Server") :: Instance
local Services = require(Server.Registry)
local Match = require(Server.Classes.Match)

local MatchService = {}

-- Services that hold per-match state. InitMatch runs in this order, CleanupMatch in
-- reverse, so e.g. enemies are removed before the crops and defenses they target.
MatchService.MatchServices = {
	"CropService",
	"SeedShopService",
	"HarvestChestService",
	"BarricadeService",
	"SentryService",
	"DefenseService",
	"EnemyService",
	"WaveService",
	"EscapeService",
}

local matches: { [number]: any } = {}
local arenaSlots: { [number]: number } = {} -- [arenaIndex] = matchId
local nextMatchId = 0

function MatchService:Init() end

function MatchService:Start()
	local step = 1 / MatchConfig.UpdateRate
	local accumulator = 0
	RunService.Heartbeat:Connect(function(dt)
		accumulator += dt
		if accumulator < step then
			return
		end
		local elapsed = accumulator
		accumulator = 0
		for _, match in matches do
			local ok, err = xpcall(match.Update, debug.traceback, match, elapsed)
			if not ok then
				warn(`[MatchService] match {match.Id} update error: {err}`)
			end
		end
	end)
end

function MatchService:_freeArena(): number?
	for index = 1, MatchConfig.MaxConcurrentMatches do
		if arenaSlots[index] == nil then
			return index
		end
	end
	return nil
end

function MatchService:CanCreateMatch(): boolean
	return self:_freeArena() ~= nil
end

function MatchService:CreateMatch(sessions: { any }): any
	local arenaIndex = self:_freeArena()
	if arenaIndex == nil or #sessions == 0 then
		return nil
	end
	nextMatchId += 1
	local matchId = nextMatchId
	arenaSlots[arenaIndex] = matchId

	local origin = MatchConfig.Arena.Origin + Vector3.new(MatchConfig.Arena.Spacing * (arenaIndex - 1), 0, 0)
	local ok, matchOrError = pcall(Match.new, matchId, arenaIndex, origin, sessions)
	if not ok then
		warn(`[MatchService] failed to build match {matchId}: {matchOrError}`)
		arenaSlots[arenaIndex] = nil
		-- Match.new may have attached sessions before failing; detach them again.
		for _, session in sessions do
			session:ResetMatchState()
		end
		return nil
	end
	local match = matchOrError
	matches[matchId] = match

	for _, serviceName in self.MatchServices do
		local service = Services[serviceName]
		local initOk, err = xpcall(service.InitMatch, debug.traceback, service, match)
		if not initOk then
			warn(`[MatchService] {serviceName}:InitMatch failed for match {matchId}: {err}`)
		end
	end

	for _, session in sessions do
		session:SetLifeState(LifeState.InMatch)
	end
	match:Start()
	return match
end

function MatchService:GetMatch(matchId: number): any
	return matches[matchId]
end

function MatchService:GetMatches(): { any }
	local list = {}
	for _, match in matches do
		table.insert(list, match)
	end
	return list
end

function MatchService:GetMatchForPlayer(player: Player): any
	local session = Services.PlayerService:GetSession(player)
	return session and session.Match
end

-- Strips match-only state from a session (coins, weapon, effects, accessories).
local function detach(session: any, match: any)
	Services.CropService:ReleasePlots(match, session.Player)
	match:DetachSession(session)
	Services.MatchCoinService:Clear(session.Player)
	Services.WeaponService:UnequipWeapon(session)
	Services.StatusEffectService:ClearTarget(session)
	session:ResetMatchState()
end

function MatchService:RemoveSession(session: any, reason: string)
	local match = session.Match
	if match == nil then
		return
	end
	Services.RewardService:CommitMatchStats(session, match, reason == "Escaped")
	detach(session, match)
	if reason ~= "Left" and not session.Destroyed then
		session:SetLifeState(LifeState.Lobby)
		task.spawn(function()
			Services.PlayerService:SpawnInLobby(session)
		end)
	end
end

function MatchService:OnPlayerRemoving(session: any)
	if session.Match then
		self:RemoveSession(session, "Left")
	end
end

function MatchService:DestroyMatch(match: any)
	if matches[match.Id] ~= match then
		return
	end
	-- Anyone still inside (match ended normally) goes back to the lobby first.
	for _, session in match:GetSessions() do
		self:RemoveSession(session, "Returned")
	end
	Services.CleanupService:CleanupMatch(match, self.MatchServices)
	matches[match.Id] = nil
	if arenaSlots[match.ArenaIndex] == match.Id then
		arenaSlots[match.ArenaIndex] = nil
	end
end

return MatchService
```

```lua
--!strict
--[[
	LobbyService
	Path: ServerScriptService.Server.Services.LobbyService

	Builds the lobby, runs the matchmaking queue and hands groups of players to
	MatchService. Players join via the Farm Portal prompt or the Play button
	(RequestJoinQueue). When the first player queues, a countdown starts; the match
	launches when it expires or the queue reaches MaxPlayersPerMatch. Private servers
	need nothing special: everyone in the server shares one queue.

	Only players with a loaded profile may queue (a failed load must never earn or
	spend persistent currency).

	Dependencies: MapBuilder, MatchConfig, LifeState, Attributes, Clock, Result,
	              ResultCode; Registry: PlayerService, MatchService, DataService,
	              RemoteService

	API:
		LobbyService:GetSpawnCFrame() -> CFrame
		LobbyService:JoinQueue(player) -> Result
		LobbyService:LeaveQueue(player) -> Result
]]

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Workspace = game:GetService("Workspace")

local Shared = ReplicatedStorage.Shared
local Attributes = require(Shared.Constants.Attributes)
local Clock = require(Shared.Utilities.Clock)
local LifeState = require(Shared.Constants.LifeState)
local MatchConfig = require(Shared.Config.MatchConfig)
local Result = require(Shared.Utilities.Result)
local ResultCode = require(Shared.Constants.ResultCode)

local Server = script:FindFirstAncestor("Server") :: Instance
local Services = require(Server.Registry)
local MapBuilder = require(Server.Systems.MapBuilder)

local LobbyService = {}

local layout: any = nil
local queue: { Player } = {}
local queueEndsAt = 0

local function publishQueue()
	for _, player in queue do
		player:SetAttribute(Attributes.QueueEndsAt, queueEndsAt)
		player:SetAttribute(Attributes.QueueSize, #queue)
	end
end

local function clearQueueAttributes(player: Player)
	player:SetAttribute(Attributes.Queued, nil)
	player:SetAttribute(Attributes.QueueEndsAt, nil)
	player:SetAttribute(Attributes.QueueSize, nil)
end

function LobbyService:Init()
	layout = MapBuilder.BuildLobby(Workspace)
end

function LobbyService:Start()
	Services.RemoteService:OnInvoke("RequestJoinQueue", function(player: Player)
		return self:JoinQueue(player)
	end)
	Services.RemoteService:OnInvoke("RequestLeaveQueue", function(player: Player)
		return self:LeaveQueue(player)
	end)
	layout.PortalPrompt.Triggered:Connect(function(player: Player)
		local session = Services.PlayerService:GetSession(player)
		if session and session.LifeState == LifeState.Queued then
			self:LeaveQueue(player)
			return
		end
		local result = self:JoinQueue(player)
		if not result.Ok then
			Services.PlayerService:Notify(player, ResultCode.Messages[result.Code] or "Can't join right now.", "Error")
		end
	end)

	task.spawn(function()
		while true do
			task.wait(0.5)
			self:_tick()
		end
	end)
end

function LobbyService:GetSpawnCFrame(): CFrame
	return layout.SpawnCFrame
end

function LobbyService:JoinQueue(player: Player)
	local session = Services.PlayerService:GetSession(player)
	if session == nil then
		return Result.Fail(ResultCode.NotAllowed)
	end
	if not Services.DataService:IsLoaded(player) then
		return Result.Fail(ResultCode.DataNotLoaded)
	end
	if session.Match ~= nil then
		return Result.Fail(ResultCode.AlreadyInMatch)
	end
	if table.find(queue, player) then
		return Result.Fail(ResultCode.AlreadyDone)
	end
	if session.LifeState ~= LifeState.Lobby then
		return Result.Fail(ResultCode.InvalidState)
	end
	table.insert(queue, player)
	session:SetLifeState(LifeState.Queued)
	player:SetAttribute(Attributes.Queued, true)
	if #queue == 1 then
		queueEndsAt = Clock.Now() + MatchConfig.QueueCountdown
	end
	publishQueue()
	return Result.Ok({ QueueSize = #queue, StartsAt = queueEndsAt })
end

function LobbyService:LeaveQueue(player: Player)
	local index = table.find(queue, player)
	if index == nil then
		return Result.Fail(ResultCode.NotFound)
	end
	table.remove(queue, index)
	clearQueueAttributes(player)
	local session = Services.PlayerService:GetSession(player)
	if session and session.LifeState == LifeState.Queued then
		session:SetLifeState(LifeState.Lobby)
	end
	publishQueue()
	return Result.Ok()
end

function LobbyService:_tick()
	-- Drop anyone who stopped being eligible (left, data released, already in a match).
	for index = #queue, 1, -1 do
		local player = queue[index]
		local session = Services.PlayerService:GetSession(player)
		if session == nil or session.Match ~= nil or session.LifeState ~= LifeState.Queued then
			table.remove(queue, index)
			clearQueueAttributes(player)
		end
	end
	if #queue == 0 then
		return
	end
	local full = #queue >= MatchConfig.MaxPlayersPerMatch
	if not full and Clock.Now() < queueEndsAt then
		return
	end
	if not Services.MatchService:CanCreateMatch() then
		-- All arenas busy: keep the queue and retry shortly.
		queueEndsAt = Clock.Now() + 5
		publishQueue()
		for _, player in queue do
			Services.PlayerService:Notify(player, ResultCode.Messages.QueueFull, "Warning")
		end
		return
	end

	local group = {}
	for _ = 1, math.min(#queue, MatchConfig.MaxPlayersPerMatch) do
		local player = table.remove(queue, 1) :: Player
		clearQueueAttributes(player)
		table.insert(group, Services.PlayerService:GetSession(player))
	end
	local match = Services.MatchService:CreateMatch(group)
	if match == nil then
		for _, session in group do
```

```lua
	CleanupService
	Path: ServerScriptService.Server.Services.CleanupService

	Guarantees nothing outlives its match. CleanupMatch tears per-match state down in
	reverse initialisation order (each service's CleanupMatch is isolated with xpcall so
	one failure cannot strand the rest), then destroys the Match (its Janitor owns the
	folder, map, core and state machine).

	A periodic sweep also removes orphans that slipped through a bug: match folders
	whose match no longer exists, and enemy models that are no longer registered.

	Dependencies: Registry: MatchService, EnemyService

	API:
		CleanupService:CleanupMatch(match, serviceNames)
		CleanupService:Sweep() -> number     -- orphans removed (also runs every 30 s)
]]

local Workspace = game:GetService("Workspace")

local Server = script:FindFirstAncestor("Server") :: Instance
local Services = require(Server.Registry)

local SWEEP_INTERVAL = 30

local CleanupService = {}

function CleanupService:Init() end

function CleanupService:Start()
	task.spawn(function()
		while true do
			task.wait(SWEEP_INTERVAL)
			local ok, result = pcall(self.Sweep, self)
			if not ok then
				warn(`[CleanupService] sweep failed: {result}`)
			elseif result > 0 then
				warn(`[CleanupService] removed {result} orphaned objects`)
			end
		end
	end)
end

function CleanupService:CleanupMatch(match: any, serviceNames: { string })
	for index = #serviceNames, 1, -1 do
		local name = serviceNames[index]
		local service = Services[name]
		local ok, err = xpcall(service.CleanupMatch, debug.traceback, service, match)
		if not ok then
			warn(`[CleanupService] {name}:CleanupMatch failed for match {match.Id}: {err}`)
		end
	end
	match:Destroy()
end

function CleanupService:Sweep(): number
	local removed = 0
	local folder = Workspace:FindFirstChild("Matches")
	if folder == nil then
		return 0
	end
	for _, matchFolder in folder:GetChildren() do
		local matchId = matchFolder:GetAttribute("MatchId")
		local match = matchId and Services.MatchService:GetMatch(matchId)
		if match == nil or match.Folder ~= matchFolder then
			matchFolder:Destroy()
			removed += 1
		else
			removed += Services.EnemyService:SweepOrphans(match)
		end
	end
	return removed
end

return CleanupService
```

```lua
		return 0
	end
	local dealt = math.min(self.Health, amount)
	self.Health -= dealt
	self:_publish()
	if self.Health <= 0 then
		self._dead = true
		self.Destroyed:Fire()
	end
	return dealt

	self.Health = math.min(self.MaxHealth, self.Health + self.MaxHealth * fraction)
	self:_publish()
```

```lua
local function buildPlots(origin: Vector3, ground: Folder, props: Folder): ({ [string]: PlotLayout }, { string })
	local farm = MatchConfig.Farm
	local plots: { [string]: PlotLayout } = {}
	local order: { string } = {}

		local angle = (index - 0.5) / farm.PlotCount * math.pi * 2
		local position = origin + Vector3.new(math.cos(angle) * farm.PlotRingRadius, 0.1, math.sin(angle) * farm.PlotRingRadius)
		local plotCFrame = CFrame.lookAt(position, Vector3.new(origin.X, position.Y, origin.Z))
		local plotId = `Plot{index}`
		local plotPart = Part({

			CanCollide = true,
			Parent = ground,
		})
		plotPart:SetAttribute(Attributes.Diggable, true)
		plotPart:SetAttribute(Attributes.PlotId, plotId)

		local slots: { SlotLayout } = {}
		local spacingX = farm.PlotSize.X / farm.SlotColumns
		local spacingZ = farm.PlotSize.Z / farm.SlotRows
		local slotIndex = 0
		for row = 1, farm.SlotRows do
			for column = 1, farm.SlotColumns do
				slotIndex += 1
				local localOffset = Vector3.new(
					(column - 0.5) * spacingX - farm.PlotSize.X / 2,
					farm.PlotSize.Y / 2,
					(row - 0.5) * spacingZ - farm.PlotSize.Z / 2
				)
				local slotPosition = plotCFrame:PointToWorldSpace(localOffset)
				local marker = Part({
					Name = `Slot{slotIndex}`,
					Size = Vector3.new(spacingX * 0.8, 0.1, spacingZ * 0.8),
					CFrame = plotCFrame.Rotation + slotPosition + Vector3.new(0, 0.05, 0),
					Color = Palette.SoilWet,
					Material = Enum.Material.Ground,
					Transparency = 0.2,
					Parent = props,
				})
				marker:SetAttribute(Attributes.PlotId, plotId)
				marker:SetAttribute(Attributes.SlotIndex, slotIndex)
				table.insert(slots, { Index = slotIndex, Position = slotPosition, Part = marker })
			end
		end

		plots[plotId] = { Id = plotId, CFrame = plotCFrame, Size = farm.PlotSize, Part = plotPart, Slots = slots }
		table.insert(order, plotId)

	return plots, order

local function buildFence(origin: Vector3, props: Folder): { Vector3 }
	local farm = MatchConfig.Farm
	local radius = farm.FenceRadius
	local segments = 56
	local gateHalfAngle = (farm.GateWidth / 2) / radius
	local gates: { Vector3 } = {}
	for gate = 0, farm.GateCount - 1 do
		local angle = gate / farm.GateCount * math.pi * 2
		table.insert(gates, origin + Vector3.new(math.cos(angle) * radius, 0, math.sin(angle) * radius))
	end

	for index = 0, segments - 1 do
		local angle = (index + 0.5) / segments * math.pi * 2
		local inGate = false
		for gate = 0, farm.GateCount - 1 do
			local gateAngle = gate / farm.GateCount * math.pi * 2
			local delta = math.abs((angle - gateAngle + math.pi) % (math.pi * 2) - math.pi)
			if delta < gateHalfAngle then
				inGate = true
				break
			end
		end
		if not inGate then
			local position = origin + Vector3.new(math.cos(angle) * radius, farm.FenceHeight / 2, math.sin(angle) * radius)
			local length = 2 * math.pi * radius / segments + 0.2
			Part({
				Name = "Fence",
				Size = Vector3.new(length, farm.FenceHeight, 0.8),
				-- lookAt points -Z at the centre, so the part's X axis runs along the ring.
				CFrame = CFrame.lookAt(position, Vector3.new(origin.X, position.Y, origin.Z)),
				Color = Palette.Wood,
				Material = Enum.Material.WoodPlanks,
				CanCollide = true,
				Parent = props,
			})
		end
	end

	for _, gatePosition in gates do
		local outward = (gatePosition - origin).Unit
		local side = Vector3.new(-outward.Z, 0, outward.X)
		for _, sign in { -1, 1 } do
			lantern(props, gatePosition + side * sign * (farm.GateWidth / 2 + 1.5))
		end
	end
	return gates
end
```

### `ServerScriptService.Server.Classes.SeedShop`

```lua
--!strict
--[[
	SeedShop
	Path: ServerScriptService.Server.Classes.SeedShop

	Per-match seed stock. Generation is a pure function of (Random, context) so it is
	deterministic for a given match seed and covered by unit tests. The purchase API is
	split into Check + Commit; SeedShopService calls both without yielding in between,
	which makes each purchase atomic under Luau's single-threaded scheduler.

	Every restock bumps Version. Clients must send the version they saw, so a purchase
	racing a restock is rejected with StockChanged instead of buying at a new price.

	Dependencies: SeedConfig, CropConfig, WeightedRandom, MathUtil, ResultCode, Clock

	API:
		SeedShop.GenerateStock(rng, context) -> { Entry }      (static, pure)
		SeedShop.new() -> SeedShop
		shop:Restock(rng, context)
		shop:CheckPurchase(userId, seedId, version) -> (boolean, code, entry?)
		shop:CommitPurchase(userId, seedId)
		shop:Snapshot(userId, open) -> ShopSnapshot
]]

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local Shared = ReplicatedStorage.Shared
local Clock = require(Shared.Utilities.Clock)
local CropConfig = require(Shared.Config.CropConfig)
local MathUtil = require(Shared.Utilities.MathUtil)
local ResultCode = require(Shared.Constants.ResultCode)
local SeedConfig = require(Shared.Config.SeedConfig)
local WeightedRandom = require(Shared.Utilities.WeightedRandom)

export type Entry = { SeedId: string, Quantity: number, Price: number, PerPlayerLimit: number? }
export type Context = { Wave: number, ContinueLevel: number, PlayerCount: number }

local SeedShop = {}
SeedShop.__index = SeedShop

local function rollEntry(seedId: string, rule: any, rng: Random, context: Context): Entry
	local crop = CropConfig.Get(seedId)
	assert(crop, `[SeedShop] seed {seedId} has no crop definition`)
	local baseQuantity = rng:NextInteger(rule.Quantity[1], rule.Quantity[2])
	local quantity = math.max(1, math.floor(baseQuantity * (1 + SeedConfig.QuantityPerExtraPlayer * (context.PlayerCount - 1))))
	local variance = rule.PriceVariance[1] + rng:NextNumber() * (rule.PriceVariance[2] - rule.PriceVariance[1])
	local price = math.max(SeedConfig.PriceRoundTo, MathUtil.RoundTo(crop.SeedPrice * variance, SeedConfig.PriceRoundTo))
	return { SeedId = seedId, Quantity = quantity, Price = price, PerPlayerLimit = rule.PerPlayerLimit }
end

function SeedShop.GenerateStock(rng: Random, context: Context): { Entry }
	local entries: { Entry } = {}
	local guaranteed: { string } = {}
	local candidates: { string } = {}

	-- Iterate in a stable order so the same seed always yields the same stock.
	local ids = {}
	for seedId in SeedConfig.Seeds do
		table.insert(ids, seedId)
	end
	table.sort(ids)

	for _, seedId in ids do
		local rule = SeedConfig.Seeds[seedId]
		if rule.AlwaysStocked then
			table.insert(guaranteed, seedId)
		elseif context.Wave >= rule.MinWave then
			table.insert(candidates, seedId)
		end
	end

	for _, seedId in guaranteed do
		table.insert(entries, rollEntry(seedId, SeedConfig.Seeds[seedId], rng, context))
	end

	local remainingSlots = math.max(0, SeedConfig.SlotCount - #entries)
	local function weightOf(seedId: string): number
		local rule = SeedConfig.Seeds[seedId]
		local crop = CropConfig.Get(seedId)
		local weight = rule.Weight
		if crop and table.find(SeedConfig.BoostedRarities, crop.Rarity) then
			weight *= 1 + SeedConfig.RarityBoostPerContinue * context.ContinueLevel
		end
		return weight
	end
	for _, seedId in WeightedRandom.PickManyUnique(candidates, remainingSlots, weightOf, rng) do
		table.insert(entries, rollEntry(seedId, SeedConfig.Seeds[seedId], rng, context))
	end

	table.sort(entries, function(a, b)
		local ca, cb = CropConfig.Get(a.SeedId), CropConfig.Get(b.SeedId)
		return (ca and ca.SortOrder or 0) < (cb and cb.SortOrder or 0)
	end)
	return entries
end

function SeedShop.new()
	return setmetatable({
		Version = 0,
		RestockedAt = 0,
		Entries = {} :: { Entry },
		_bySeed = {} :: { [string]: Entry },
		_purchases = {} :: { [number]: { [string]: number } },
	}, SeedShop)
end

function SeedShop:Restock(rng: Random, context: Context)
	self.Version += 1
	self.RestockedAt = Clock.Now()
	self.Entries = SeedShop.GenerateStock(rng, context)
	self._bySeed = {}
	for _, entry in self.Entries do
		self._bySeed[entry.SeedId] = entry
	end
	self._purchases = {}
end

function SeedShop:GetPurchased(userId: number, seedId: string): number
	local perPlayer = self._purchases[userId]
	return perPlayer and perPlayer[seedId] or 0
end

function SeedShop:CheckPurchase(userId: number, seedId: string, version: number): (boolean, string, Entry?)
	if version ~= self.Version then
		return false, ResultCode.StockChanged, nil
	end
	local entry = self._bySeed[seedId]
	if entry == nil then
		return false, ResultCode.NotFound, nil
	end
	if entry.Quantity <= 0 then
		return false, ResultCode.OutOfStock, nil
	end
	if entry.PerPlayerLimit and self:GetPurchased(userId, seedId) >= entry.PerPlayerLimit then
		return false, ResultCode.PurchaseLimit, nil
	end
	return true, ResultCode.Ok, entry
end

function SeedShop:CommitPurchase(userId: number, seedId: string)
	local entry = self._bySeed[seedId]
	assert(entry and entry.Quantity > 0, "[SeedShop] CommitPurchase without a valid CheckPurchase")
	entry.Quantity -= 1
	local perPlayer = self._purchases[userId]
	if perPlayer == nil then
		perPlayer = {}
		self._purchases[userId] = perPlayer
	end
	perPlayer[seedId] = (perPlayer[seedId] or 0) + 1
end

function SeedShop:Snapshot(userId: number, open: boolean)
	local entries = {}
	for _, entry in self.Entries do
		table.insert(entries, {
			SeedId = entry.SeedId,
			Quantity = entry.Quantity,
			Price = entry.Price,
			PerPlayerLimit = entry.PerPlayerLimit,
			Purchased = self:GetPurchased(userId, entry.SeedId),
		})
	end
	return { Version = self.Version, Open = open, Entries = entries, RestockedAt = self.RestockedAt }
end

return SeedShop
```

```lua
--!strict
--[[
	SeedShopService
	Path: ServerScriptService.Server.Services.SeedShopService

	Runs the randomized Seed Shop of every match. Stock is rolled on the server from
	SeedConfig with the match's seeded Random at each Intermission and each
	ContinuePreparation, then broadcast (ShopUpdated). Purchase flow:

		client RequestBuySeed(seedId, stockVersion)
		-> session/match/profile validation -> shop open in this state
		-> stock + version + per-player limit check (SeedShop:CheckPurchase)
		-> Candies check and charge + seed grant in ONE profile mutation
		-> stock decrement (SeedShop:CommitPurchase)
		-> response + ShopUpdated broadcast to the match

	Nothing between the stock check and the decrement yields, so two simultaneous
	requests for the last seed can never both succeed.

	Dependencies: SeedShop, Result, ResultCode; Registry: PlayerService, DataService,
	              RemoteService

	API:
		SeedShopService:InitMatch(match) / CleanupMatch(match)
		SeedShopService:Restock(match)
		SeedShopService:SetOpen(match, open)
		SeedShopService:IsOpen(match) -> boolean
		SeedShopService:Purchase(player, seedId, version) -> Result
		SeedShopService:GetSnapshot(match, player) -> ShopSnapshot?
]]

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local Shared = ReplicatedStorage.Shared
local Result = require(Shared.Utilities.Result)
local ResultCode = require(Shared.Constants.ResultCode)

local Server = script:FindFirstAncestor("Server") :: Instance
local Services = require(Server.Registry)
local SeedShop = require(Server.Classes.SeedShop)

local SeedShopService = {}

local states: { [any]: { Shop: any, Open: boolean } } = {}

function SeedShopService:Init() end

function SeedShopService:Start()
	Services.RemoteService:OnInvoke("RequestBuySeed", function(player: Player, seedId: string, version: number)
		return self:Purchase(player, seedId, version)
	end)
end

function SeedShopService:InitMatch(match: any)
	states[match] = { Shop = SeedShop.new(), Open = false }
end

function SeedShopService:CleanupMatch(match: any)
	states[match] = nil
end

function SeedShopService:GetSnapshot(match: any, player: Player): any
	local state = states[match]
	if state == nil then
		return nil
	end
	return state.Shop:Snapshot(player.UserId, state.Open)
end

function SeedShopService:_broadcast(match: any)
	for _, player in match:GetPlayers() do
		Services.RemoteService:FireClient("ShopUpdated", player, self:GetSnapshot(match, player))
	end
end

function SeedShopService:Restock(match: any)
	local state = states[match]
	if state == nil then
		return
	end
	local context = match:GetContext()
	-- Seeds unlock by completed waves (MinWave), so use the raw wave count here.
	context.Wave = match.Wave
	state.Shop:Restock(match.Rng, context)
	self:_broadcast(match)
	match:FireAll("MatchEvent", "ShopRestocked", { Version = state.Shop.Version })
end

function SeedShopService:SetOpen(match: any, open: boolean)
	local state = states[match]
	if state == nil or state.Open == open then
		return
	end
	state.Open = open
	self:_broadcast(match)
end

function SeedShopService:IsOpen(match: any): boolean
	local state = states[match]
	return state ~= nil and state.Open
end

function SeedShopService:Purchase(player: Player, seedId: string, version: number)
	local session = Services.PlayerService:GetSession(player)
	local match = session and session.Match
	if match == nil then
		return Result.Fail(ResultCode.NotInMatch)
	end
	local state = states[match]
	if state == nil then
		return Result.Fail(ResultCode.NotInMatch)
	end
	if not Services.DataService:IsLoaded(player) then
		return Result.Fail(ResultCode.DataNotLoaded)
	end
	if not state.Open then
		return Result.Fail(ResultCode.InvalidState)
	end

	local ok, code, entry = state.Shop:CheckPurchase(player.UserId, seedId, version)
	if not ok or entry == nil then
		return Result.Fail(code)
	end

	local price = entry.Price
	local mutated, paid = Services.DataService:Update(player, function(data)
		if data.Candies < price then
			return false
		end
		data.Candies -= price
		data.Seeds[seedId] = (data.Seeds[seedId] or 0) + 1
		return true
	end)
	if not mutated then
		return Result.Fail(ResultCode.ServerError)
	end
	if not paid then
		return Result.Fail(ResultCode.InsufficientFunds)
	end

	state.Shop:CommitPurchase(player.UserId, seedId)
	self:_broadcast(match)
	return Result.Ok({ SeedId = seedId, Remaining = entry.Quantity })
end

return SeedShopService
```

```lua
function Crop:_currentStageIndex(): number
	local index = 1
	for i, stage in self.Definition.Stages do
		if self.Progress >= stage.AtProgress then
			index = i
		end
	end
	return index

-- Rebuilds the visual when the growth stage changes (4 rebuilds over a crop's life).
function Crop:_refreshStage()
	local index = self:_currentStageIndex()
	if index == self.StageIndex then
		return
	end
	self.StageIndex = index
	local stage = self.Definition.Stages[index]
	local ok, visualOrError = pcall(ModelFactory.CreateCropVisual, self.Definition, stage.Scale)
	if not ok then
		warn(`[Crop] visual for {self.CropId} failed: {visualOrError}`)
		return
	end
	local visual = visualOrError :: Model
	visual:PivotTo(CFrame.new(self.Position))
	local previous = self.Model:FindFirstChild("Visual")
	if previous then
		previous:Destroy()
	end
	visual.Parent = self.Model
	self.Model:SetAttribute(Attributes.Stage, stage.Name)
end

	local model = self.Model
	-- Growth replicates in 1% steps so attribute traffic stays tiny.
	local growth = math.floor(self.Progress * 100) / 100
	if growth ~= self._publishedGrowth then
		self._publishedGrowth = growth
		model:SetAttribute(Attributes.Growth, growth)
	end
	model:SetAttribute(Attributes.Health, math.ceil(self.Health))
	model:SetAttribute(Attributes.MaxHealth, self.MaxHealth)
	model:SetAttribute(Attributes.Ripe, self:IsRipe())
end

function Crop:Update(dt: number)
	if self.Removed then
		return
	end
	local changed = false
	if self.Progress < 1 then
		self.Progress = math.min(1, self.Progress + dt * self.GrowthRate)
		self:_refreshStage()
		changed = true
	end
	local special = self.Definition.Special
	if special and special.Type == "Regenerate" and self.Health < self.MaxHealth then
		self.Health = math.min(self.MaxHealth, self.Health + (special.HealthPerSecond or 0) * dt)
		changed = true
	end
	if changed then
		self:_publish()
	end
end

function Crop:IsRipe(): boolean
	return self.Progress >= 1
end

function Crop:IsAlive(): boolean
	return not self.Removed and self.Health > 0
end

function Crop:GetPosition(): Vector3
	return self.Position
end

function Crop:GetTargetPriorityBonus(): number
	local special = self.Definition.Special
	if special and special.Type == "Decoy" then
		return special.PriorityBonus or 0
	end
	return 0
end

function Crop:CanBeStolen(): boolean
	local special = self.Definition.Special
	return not (special and special.Type == "Anchored")
end

function Crop:TakeDamage(amount: number, source: any): number
	if not self:IsAlive() or amount <= 0 then
		return 0
	end
	local movementType = source and source.MovementType or "Ground"
	if source and source.IsBoss then
		movementType = "Boss"
	end
	local vulnerability = (self.Definition.Vulnerability :: any)[movementType] or 1
	local dealt = math.min(self.Health, amount * vulnerability)
	self.Health -= dealt
	self:_publish()

	local special = self.Definition.Special
	if special and special.Type == "Thorns" and source and source.Enemy and source.Enemy:IsAlive() then
		Services.DamageService:DamageEnemy(source.Enemy, special.Damage or 0, { Kind = "Crop", Id = self.CropId })
	end

	if self.Health <= 0 then
		self._onDepleted(self)
	end
	return dealt
end

function Crop:Heal(fraction: number)
	if not self:IsAlive() then
		return
	end
	self.Health = math.min(self.MaxHealth, self.Health + self.MaxHealth * fraction)
	self:_publish()
end

function Crop:ForceRipen()
	self.Progress = 1
	self:_refreshStage()
	self:_publish()
end

function Crop:Destroy()
	if self.Removed then
		return
	end
	self.Removed = true
	self.Janitor:Destroy()
end

return Crop
```

```lua
	DisplayName: string,
	Icon: string,

	UnitValue: number,
	Value: number,

function HarvestChest.new(capacity: Capacity?)
	return setmetatable({
		_entries = {} :: { [string]: ChestEntry },
		_forfeited = false,
		_capacity = capacity or { MaxItems = math.huge, MaxStacks = math.huge },
		_items = 0,
		_stacks = 0,
	}, HarvestChest)
end

function HarvestChest:CanDeposit(itemId: string, quantity: number): boolean
	if self._forfeited then
		return false
	end
	if self._items + quantity > self._capacity.MaxItems then
		return false
	end
	if self._entries[itemId] == nil and self._stacks >= self._capacity.MaxStacks then
		return false
	end
	return true
end

function HarvestChest:Deposit(itemId: string, displayName: string, icon: string, quantity: number, unitValue: number, force: boolean?): boolean
	if self._forfeited then
		return false
	end
	if type(quantity) ~= "number" or quantity <= 0 or quantity % 1 ~= 0 then
		return false
	end
	if type(unitValue) ~= "number" or unitValue < 0 or unitValue ~= unitValue then
		return false
	end
	if not force and not self:CanDeposit(itemId, quantity) then
		return false
	end
	local entry = self._entries[itemId]
	if entry == nil then
		entry = { ItemId = itemId, DisplayName = displayName, Icon = icon, Quantity = 0, UnitValue = unitValue, Value = 0 }
		self._entries[itemId] = entry
		self._stacks += 1
	end
	self._items += quantity
	entry.Quantity += quantity
	entry.UnitValue = unitValue
	entry.Value += quantity * unitValue
	return true
end

function HarvestChest:GetTotalValue(): number
	local total = 0
	for _, entry in self._entries do
		total += entry.Value
	end
	return total
end

function HarvestChest:GetItemCount(): number
	return self._items
end

function HarvestChest:GetCapacity(): { Items: number, MaxItems: number, Stacks: number, MaxStacks: number }
	return {
		Items = self._items,
		MaxItems = self._capacity.MaxItems,
		Stacks = self._stacks,
		MaxStacks = self._capacity.MaxStacks,
	}
end

function HarvestChest:GetEntries(): { ChestEntry }
	local list = {}
	for _, entry in self._entries do
		table.insert(list, table.clone(entry))
	end
	table.sort(list, function(a, b)
		if a.Value == b.Value then
			return a.ItemId < b.ItemId
		end
		return a.Value > b.Value
	end)
	return list
end

function HarvestChest:ComputePayout(conversionRate: number, riskBonus: number): number
	if self._forfeited then
		return 0
	end
	return math.floor(self:GetTotalValue() * conversionRate * (1 + riskBonus))
end

	self._forfeited = true
	table.clear(self._entries)
	self._items = 0
	self._stacks = 0
end

	return self._forfeited
end
```

```lua
		return
```

```lua
	Path: ServerScriptService.Server.Services.EscapeService

	Runs the Extraction decision: the Escape Door, each player's one-time choice, the
	payout, and the resolution that tells the Match what happens next.

	Rules:
	  * only players in the match, alive, within DoorRange of the door may decide
	  * each player decides exactly once (Decisions[userId]); repeats -> AlreadyDone
	  * ESCAPE pays immediately and returns the player to the lobby
	  * when everyone has decided, or the window expires:

	      - nobody chose CONTINUE -> undecided players are escaped (paid) -> "End"

	   1. mark Paid[userId] BEFORE anything else (a duplicate can never pay twice)
	   2. freeze the payout from the server-owned chest (value x rate x risk bonus)
	   3. add Candies to the profile (CurrencyService -> CandyService -> DataService)
	   4. record statistics (RewardService.CommitMatchStats, escaped = true)
	   5. save the profile
	   6. strip match-only state (coins, weapon, effects, accessories)
	   7. return the player to the lobby (MatchService.RemoveSession)

	              Format; Registry: PlayerService, MatchService, HarvestChestService,

local Services = require(Server.Registry)
local EscapeDoor = require(Server.Classes.EscapeDoor)

	Open: boolean,
	Door: any,
	Decisions: { [number]: string },
	Paid: { [number]: boolean },
}

local EscapeService = {}

local states: { [any]: ExtractionState } = {}

function EscapeService:Init() end

function EscapeService:Start()

		return self:SubmitDecision(player, "Escape")

	for userId, decision in state.Decisions do
		decisions[tostring(userId)] = decision
	end

	return {
		Open = true,
		DeadlineAt = match.PhaseEndsAt,
		Decisions = decisions,
		YourDecision = state.Decisions[player.UserId],

	end)
	local nextDanger = Services.DifficultyService:GetDangerMultiplier(match.ContinueLevel + 1)
	state.Door:SetPromptText(Services.HarvestChestService:ComputePayout(match), nextDanger)

function EscapeService:SubmitDecision(player: Player, decision: string)
	if decision ~= "Escape" and decision ~= "Continue" then
		return Result.Fail(ResultCode.InvalidPayload)
	end
	local session = Services.PlayerService:GetSession(player)
	local match = session and session.Match
	if match == nil then
		return Result.Fail(ResultCode.NotInMatch)
	end
	local state = states[match]
	if state == nil or not state.Open then
		return Result.Fail(ResultCode.InvalidState)
	end
	if session.LifeState ~= LifeState.InMatch or not session:IsAlive() then
		return Result.Fail(ResultCode.Dead)
	end
	if state.Decisions[player.UserId] ~= nil or state.Paid[player.UserId] then
		return Result.Fail(ResultCode.AlreadyDone)
	end
	local position = session:GetPosition()
	if position == nil or state.Door == nil or (position - state.Door:GetPosition()).Magnitude > MatchConfig.Interaction.DoorRange then
		return Result.Fail(ResultCode.OutOfRange)
	end

	state.Decisions[player.UserId] = decision
	if decision == "Escape" then
		self:_processEscape(match, session)
	else
		match:Notify(`{player.DisplayName} chose to CONTINUE!`, "Warning")
	end
	self:_broadcast(match)
	return Result.Ok({ Decision = decision })
end

function EscapeService:_processEscape(match: any, session: any)
	local state = states[match]
	local player: Player = session.Player
	if state == nil or state.Paid[player.UserId] then
		return
	end
	state.Paid[player.UserId] = true
	state.Decisions[player.UserId] = "Escape"

	local payout = Services.HarvestChestService:ComputePayout(match)
	local paid = false
	if payout > 0 then
		paid = Services.CurrencyService:Grant(player, CurrencyId.Candies, payout, "Extraction")
		if not paid then
			warn(`[EscapeService] payout of {payout} to {player.UserId} failed (profile not loaded?)`)
		end
	end
	Services.MatchService:RemoveSession(session, "Escaped")

	local message = if paid
		then `You escaped with {Format.Commas(payout)} Candies!`
		else "You escaped! (The chest was empty, so no Candies this time.)"
	Services.PlayerService:Notify(player, message, "Success", 6)
	Services.RemoteService:FireClient("MatchEvent", player, "Escaped", { Candies = if paid then payout else 0, Wave = match.Wave })
	match:Notify(`{player.DisplayName} escaped to the lobby.`, "Info")
end

function EscapeService:Update(match: any, deadlinePassed: boolean): string?
	local state = states[match]
	if state == nil or not state.Open then
		return nil
	end
	local sessions = match:GetSessions()
	if #sessions == 0 then
		return "End"
	end

	local anyContinue = false
	local allDecided = true
	for _, session in sessions do
		local decision = state.Decisions[session.UserId]
		if decision == "Continue" then
			anyContinue = true
		elseif decision == nil then
			allDecided = false
		end
	end
	if not allDecided and not deadlinePassed then
		return nil
	end

	if anyContinue then
		for _, session in sessions do
			if state.Decisions[session.UserId] == nil then
				state.Decisions[session.UserId] = "Continue"
				Services.PlayerService:Notify(session.Player, "You stayed on the farm. Get ready!", "Warning")
			end
		end
		return "Continue"
	end

	-- Nobody chose to continue: escape everyone still here so their haul is banked.
	for _, session in sessions do
		if not state.Paid[session.UserId] then
			self:_processEscape(match, session)
		end
	end
	return "End"
end

return EscapeService
```

```lua
	setmetatable(self, Sentry)
	self.NextFireAt = 0
	self.Head = self.Model:FindFirstChild("Head") :: BasePart?

	return self.Definition.Range * self:GetMultiplier("RangeMultiplier")
end

function Sentry:GetDamage(): number
	return self.Definition.Damage * self:GetMultiplier("DamageMultiplier")
end

function Sentry:_source(): any
	local owner = self.Owner
	return {
		Kind = "Sentry",
		Id = self.DefenseId,
		Player = if owner and self.Match.Sessions[owner] then owner else nil,
	}
end

function Sentry:_canSee(position: Vector3): boolean
	local from = self.MuzzlePosition
	return Workspace:Raycast(from, position - from, self._sightParams) == nil
end

function Sentry:_aimAt(position: Vector3)
	local head = self.Head
	if head then
		local headPosition = head.CFrame.Position
		local flat = Vector3.new(position.X, headPosition.Y, position.Z)
		if (flat - headPosition).Magnitude > 0.1 then
			head.CFrame = CFrame.lookAt(headPosition, flat)
		end
	end
end

function Sentry:Update(now: number)
	if not self:IsAlive() or now < self.NextFireAt then
		return
	end
	local definition = self.Definition
	local match = self.Match
	local range = self:GetRange()

	if definition.SentryType == "Pulse" then
		local hit = Services.DamageService:DamageEnemiesInRadius(match, self.Position, range, self:GetDamage(), self:_source(), {
			IncludeUnderground = false,
			IncludeAir = definition.CanTargetAir,
			Effect = definition.Effect,
		})
		if #hit > 0 then
			self.NextFireAt = now + definition.FireInterval
			Services.EffectService:PlayAt(match, "LanternPulse", self.Position, { Radius = range })
		end
		return
	end

	local target = Services.EnemyService:GetNearestEnemy(match, self.Position, range, function(enemy)
		if enemy.Underground or (enemy.IsAir and not definition.CanTargetAir) then
			return false
		end
		return self:_canSee(enemy:GetPosition())
	end)
	if target == nil then
		return
	end
	self.NextFireAt = now + definition.FireInterval
	local targetPosition = target:GetPosition()
	self:_aimAt(targetPosition)
	Services.DamageService:DamageEnemy(target, self:GetDamage(), self:_source())
	Services.EffectService:PlayAt(match, "SentryBolt", self.MuzzlePosition, {
		From = self.MuzzlePosition,
		To = targetPosition,
		Color = definition.Color,
	})
end

	local head = self.Head
	if head then
		head.Size *= 1.08
		self.MuzzlePosition = head.CFrame.Position
	end
end
```

```lua
-- Placement --------------------------------------------------------------------------------

function DefenseService:ValidatePlacement(match: any, session: any, definition: any, position: Vector3, rotationY: number): (boolean, string, CFrame?)
	local placement = DefenseConfig.Placement
	local arena = match.Arena
	local size: Vector3 = definition.Size

	local fromCenter = MathUtil.FlatDistance(position, arena.Center)
	if fromCenter < placement.InnerRadius or fromCenter > placement.OuterRadius then
		return false, ResultCode.Blocked, nil
	end
	local footprint = math.max(size.X, size.Z) / 2
	local bounds = arena.Bounds
	if position.X - footprint < bounds.MinX or position.X + footprint > bounds.MaxX
		or position.Z - footprint < bounds.MinZ or position.Z + footprint > bounds.MaxZ then
		return false, ResultCode.Blocked, nil
	end

	local playerPosition = session:GetPosition()
	if playerPosition == nil or MathUtil.FlatDistance(playerPosition, position) > placement.MaxPlaceDistance then
		return false, ResultCode.OutOfRange, nil
	end

	local groundParams = RaycastParams.new()
	groundParams.FilterType = Enum.RaycastFilterType.Include
	groundParams.FilterDescendantsInstances = { arena.GroundFolder }
	local hit = Workspace:Raycast(Vector3.new(position.X, position.Y + 20, position.Z), Vector3.new(0, -60, 0), groundParams)
	if hit == nil or math.abs(hit.Position.Y - position.Y) > placement.MaxGroundSnap then
		return false, ResultCode.Blocked, nil
	end
	local floorPart = hit.Instance
	if floorPart:GetAttribute(Attributes.Diggable) ~= true or floorPart:GetAttribute(Attributes.PlotId) ~= nil then
		return false, ResultCode.Blocked, nil
	end

	local rotation = CFrame.Angles(0, math.rad(rotationY), 0)
	local cframe = CFrame.new(hit.Position) * rotation
	local overlapParams = OverlapParams.new()
	overlapParams.FilterType = Enum.RaycastFilterType.Include
	overlapParams.FilterDescendantsInstances = {
		arena.GroundFolder,
		arena.PropsFolder,
		match.Folder:FindFirstChild("Defenses") :: Instance,
		match.Folder:FindFirstChild("Crops") :: Instance,
		match.Core.Model,
	}
	local clearance = placement.Clearance
	-- The box starts just above the floor so the floor itself never counts.
	local boxSize = Vector3.new(size.X + clearance * 2, size.Y - 0.3, size.Z + clearance * 2)
	local boxCFrame = cframe * CFrame.new(0, 0.3 + boxSize.Y / 2, 0)
	local overlaps = Workspace:GetPartBoundsInBox(boxCFrame, boxSize, overlapParams)
	for _, part in overlaps do
		-- Slot markers are thin decals on plots; plots themselves are caught via the
		-- ground folder, so markers would only duplicate that result.
		if part:GetAttribute(Attributes.SlotIndex) == nil then
			return false, ResultCode.Blocked, nil
		end
	end

	local defenses = self:GetDefenses(match)
	if #defenses >= placement.MaxDefensesPerMatch then
		return false, ResultCode.LimitReached, nil
	end
	if definition.Kind == "Sentry" then
		local sentries = 0
		for _, defense in defenses do
			if defense.Kind == "Sentry" then
				sentries += 1
			end
		end
		if sentries >= placement.MaxSentriesPerMatch then
			return false, ResultCode.LimitReached, nil
		end
	end
	return true, ResultCode.Ok, cframe
end

function DefenseService:PurchaseDefense(player: Player, defenseId: string, position: Vector3, rotationY: number)

	if session == nil then
		return Result.Fail(problem :: string)
	end
	local definition = DefenseConfig.Defenses[defenseId]
	if definition == nil then
		return Result.Fail(ResultCode.NotFound)
	end
	local match = session.Match
	local state = states[match]
	if state == nil then
		return Result.Fail(ResultCode.NotInMatch)
	end
	if not Services.CurrencyService:CanAfford(player, CurrencyId.MatchCoins, definition.Cost) then
		return Result.Fail(ResultCode.InsufficientFunds)
	end
	local valid, code, cframe = self:ValidatePlacement(match, session, definition, position, rotationY)
	if not valid or cframe == nil then
		return Result.Fail(code)
	end
	if not Services.CurrencyService:TrySpend(player, CurrencyId.MatchCoins, definition.Cost, "Defense:" .. defenseId) then
		return Result.Fail(ResultCode.InsufficientFunds)
	end

	local onDepleted = function(defense: any)
		self:DestroyDefense(defense, "Destroyed")
	end
	local ok, defenseOrError = pcall(function()
		if definition.Kind == "Sentry" then
			return Services.SentryService:Create(match, definition, cframe, player, onDepleted)
		end
		return Services.BarricadeService:Create(match, definition, cframe, player, onDepleted)
	end)
	if not ok then
		warn(`[DefenseService] failed to build {defenseId}: {defenseOrError}`)
		Services.CurrencyService:Grant(player, CurrencyId.MatchCoins, definition.Cost, "Refund")
		return Result.Fail(ResultCode.ServerError)
	end
	state.Index[defenseOrError.Uid] = defenseOrError
	Services.EffectService:PlayAt(match, "Build", cframe.Position, { DefenseId = defenseId })
	return Result.Ok({ DefenseUid = defenseOrError.Uid })
end

	if session == nil then
		return Result.Fail(problem :: string)

	local defense = self:GetDefense(session.Match, defenseUid)
	if defense == nil or not defense:IsAlive() then
		return Result.Fail(ResultCode.NotFound)
	end
	local upgrade = defense:GetNextUpgrade()
	if upgrade == nil then
		return Result.Fail(ResultCode.MaxLevel)
	end
	local playerPosition = session:GetPosition()
	if playerPosition == nil or (playerPosition - defense:GetPosition()).Magnitude > DefenseConfig.Placement.MaxPlaceDistance then
		return Result.Fail(ResultCode.OutOfRange)
	end
	if not Services.CurrencyService:TrySpend(player, CurrencyId.MatchCoins, upgrade.Cost, "Upgrade:" .. defense.DefenseId) then
		return Result.Fail(ResultCode.InsufficientFunds)
	end
	defense:ApplyUpgrade()
	Services.EffectService:PlayAt(session.Match, "Repair", defense:GetPosition(), {})
	return Result.Ok({ Level = defense.Level })
end

-- Items ----------------------------------------------------------------------------------------

function DefenseService:PurchaseItem(player: Player, itemId: string)
	local session, problem = activeSession(player)
	if session == nil then
		return Result.Fail(problem :: string)
	end
	local item = DefenseConfig.Items[itemId]
	if item == nil then
		return Result.Fail(ResultCode.NotFound)
	end
	local match = session.Match
	local now = os.clock()

	if item.Kind == "Accessory" then
		if session.Accessories[itemId] then
			return Result.Fail(ResultCode.AlreadyOwned)
		end
		if not Services.CurrencyService:TrySpend(player, CurrencyId.MatchCoins, item.Cost, "Accessory:" .. itemId) then
			return Result.Fail(ResultCode.InsufficientFunds)
		end
		session.Accessories[itemId] = true
		session:RecomputeStats()
		return Result.Ok({ ItemId = itemId })
	end

	local readyAt = session.UtilityReadyAt[itemId]
	if readyAt and now < readyAt then
		return Result.Fail(ResultCode.Cooldown)
	end

	if itemId == "HealingTonic" then
		local humanoid = session.Humanoid
		if humanoid == nil or humanoid.Health >= humanoid.MaxHealth then
			return Result.Fail(ResultCode.AlreadyDone)
		end
		if not Services.CurrencyService:TrySpend(player, CurrencyId.MatchCoins, item.Cost, "Utility:" .. itemId) then
			return Result.Fail(ResultCode.InsufficientFunds)
		end
		session:Heal(item.HealAmount)
		Services.EffectService:PlayAt(match, "Heal", session:GetPosition() or match.Arena.Center, {})
	elseif itemId == "RepairKit" then
		local damaged = false
		for _, defense in self:GetDefenses(match) do
			if defense.Health < defense.MaxHealth then
				damaged = true
				break
			end
		end
		if not damaged then
			return Result.Fail(ResultCode.AlreadyDone)
		end
		if not Services.CurrencyService:TrySpend(player, CurrencyId.MatchCoins, item.Cost, "Utility:" .. itemId) then
			return Result.Fail(ResultCode.InsufficientFunds)
		end
		self:RepairAll(match, item.RepairFraction)
	else
		return Result.Fail(ResultCode.NotFound)
	end
	session.UtilityReadyAt[itemId] = now + (item.Cooldown or 0)
	return Result.Ok({ ItemId = itemId })
end
```

```lua
-- Generic part constructor with sensible defaults for decorative geometry.
function ModelFactory.Part(props: { [string]: any }): Part
	local part = Instance.new("Part")
	part.Anchored = true
	part.CanCollide = false
	part.CanTouch = false
	part.Material = Enum.Material.SmoothPlastic
	part.TopSurface = Enum.SurfaceType.Smooth
	part.BottomSurface = Enum.SurfaceType.Smooth
	for key, value in props do
		(part :: any)[key] = value
	end
	return part
end

local function ellipsoid(props: { [string]: any }): Part
	local part = ModelFactory.Part(props)
	local mesh = Instance.new("SpecialMesh")
	mesh.MeshType = Enum.MeshType.Sphere
	mesh.Parent = part
	return part
end

local function wedge(props: { [string]: any }): WedgePart
	local part = Instance.new("WedgePart")
	part.Anchored = true
	part.CanCollide = false
	part.CanTouch = false
	part.Material = Enum.Material.SmoothPlastic
	for key, value in props do
		(part :: any)[key] = value
	end
	return part
end

local function weld(part0: BasePart, part1: BasePart)
	local constraint = Instance.new("WeldConstraint")
	constraint.Part0 = part0
	constraint.Part1 = part1
	constraint.Parent = part1
end

local function findAsset(folderName: string, assetName: string): Instance?
	local assets = ReplicatedStorage:FindFirstChild("Assets")
	local folder = assets and assets:FindFirstChild(folderName)
	return folder and folder:FindFirstChild(assetName)
end

-- Makes every part of a procedurally built rig unanchored, welded to the root and
-- massless (except the root, whose mass the Humanoid needs).
local function rigToRoot(model: Model, root: BasePart)
	for _, descendant in model:GetDescendants() do
		if descendant:IsA("BasePart") and descendant ~= root then
			weld(root, descendant)
			descendant.Anchored = false
			descendant.Massless = true
			descendant.CanCollide = false
		end
	end
	root.Anchored = false
end

-- A simple humanoid-shaped rig standing on y = 0 with the HumanoidRootPart at hip
-- height. Arms can be posed forward (zombie stance).
local function buildBiped(name: string, visual: any, armsForward: boolean): RigParts
	local s = visual.Scale or 1
	local model = Instance.new("Model")
	model.Name = name

	local legHeight = 2 * s
	local root = ModelFactory.Part({
		Name = "HumanoidRootPart",
		Size = Vector3.new(2, 2, 1) * s,
		CFrame = CFrame.new(0, legHeight + s, 0),
		Transparency = 1,
		CanCollide = true,
		Parent = model,
	})

	local torso = ModelFactory.Part({
		Name = "Torso",
		Size = Vector3.new(2, 2, 1) * s,
		CFrame = root.CFrame,
		Color = visual.ShirtColor,
		Parent = model,
	})
	local head = ModelFactory.Part({
		Name = "Head",
		Size = Vector3.new(1.4, 1.4, 1.4) * s,
		CFrame = root.CFrame * CFrame.new(0, 1.75 * s, 0),
		Color = visual.SkinColor,
		Parent = model,
	})
	for side = -1, 1, 2 do
		ModelFactory.Part({
			Name = if side < 0 then "LeftLeg" else "RightLeg",
			Size = Vector3.new(0.9, 2, 0.9) * s,
			CFrame = CFrame.new(0.5 * side * s, legHeight / 2, 0),
			Color = visual.PantsColor,
			Parent = model,
		})
		-- Glowing eyes keep enemies readable at a distance.
		ModelFactory.Part({
			Name = "Eye",
			Size = Vector3.new(0.25, 0.25, 0.1) * s,
			CFrame = head.CFrame * CFrame.new(0.3 * side * s, 0.15 * s, -0.71 * s),
			Color = Color3.fromRGB(255, 240, 120),
			Material = Enum.Material.Neon,
			Parent = model,
		})
	end

	local armOffset = if armsForward
		then CFrame.new(0, 0.6 * s, -0.9 * s) * CFrame.Angles(math.rad(90), 0, 0)
		else CFrame.new(0, 0, 0)
	local leftArm = ModelFactory.Part({
		Name = "LeftArm",
		Size = Vector3.new(0.8, 2, 0.8) * s,
		CFrame = root.CFrame * CFrame.new(-1.4 * s, 0, 0) * armOffset,
		Color = visual.SkinColor,
		Parent = model,
	})
	local rightArm = ModelFactory.Part({
		Name = "RightArm",
		Size = Vector3.new(0.8, 2, 0.8) * s,
		CFrame = root.CFrame * CFrame.new(1.4 * s, 0, 0) * armOffset,
		Color = visual.SkinColor,
		Parent = model,
	})

	return {
		Model = model,
		Root = root,
		Torso = torso,
		Head = head,
		LeftArm = leftArm,
		RightArm = rightArm,
		Scale = s,
	}
end

local function addHumanoid(rig: RigParts)
	local humanoid = Instance.new("Humanoid")
	-- R15 rig type makes the Humanoid honour HipHeight on a custom (jointless) rig;
	-- R6 would look for leg parts to stand on instead.
	humanoid.RigType = Enum.HumanoidRigType.R15
	humanoid.HipHeight = 2 * rig.Scale
	humanoid.RequiresNeck = false
	humanoid.BreakJointsOnDeath = false
	humanoid.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
	humanoid.HealthDisplayType = Enum.HumanoidHealthDisplayType.AlwaysOff
	humanoid.AutomaticScalingEnabled = false
	humanoid.Parent = rig.Model
end

function ModelFactory.CreateEnemy(definition: any): Model
	local template = findAsset("Enemies", definition.Id)
	if template and template:IsA("Model") and template.PrimaryPart then
		local clone = template:Clone()
		clone.Name = definition.Id
		return clone
	end

	local visual = definition.Visual
	local kind = visual.Kind
	if kind == "Crow" then
		return buildCrow(definition.Id, visual)
	end

	local rig = buildBiped(definition.Id, visual, kind == "Zombie" or kind == "Mole")
	if kind == "Zombie" then
		decorateZombie(rig, visual)
	elseif kind == "Mole" then
		decorateMole(rig, visual)
	elseif kind == "Scarecrow" then
		decorateScarecrow(rig, visual)
	elseif kind == "Witch" then
		decorateWitch(rig, visual)
	elseif kind == "Harvester" then
		decorateHarvester(rig, visual)
	end
	rig.Model.PrimaryPart = rig.Root
	rigToRoot(rig.Model, rig.Root)
	addHumanoid(rig)
	return rig.Model
end

-- Builds the plant visual at `scale` (0..1 of ripe size). The model's pivot sits on
-- the soil surface so callers can PivotTo the slot position directly.
function ModelFactory.CreateCropVisual(definition: any, scale: number): Model
	local template = findAsset("Crops", definition.Id)
	if template and template:IsA("Model") and template.PrimaryPart then
		local clone = template:Clone()
		clone.Name = "Visual"
		for _, d in clone:GetDescendants() do
			if d:IsA("BasePart") then
				d.Anchored = true
				d.CanCollide = false
			end
		end
		if scale > 0 and scale ~= 1 then
			clone:ScaleTo(scale)
		end
		return clone
	end

	local visual = definition.Visual
	local size = visual.RipeSize * math.max(scale, 0.15)
	local model = Instance.new("Model")
	model.Name = "Visual"

	local pivot = ModelFactory.Part({
		Name = "Pivot",
		Size = Vector3.new(0.2, 0.2, 0.2),
		CFrame = CFrame.new(0, 0, 0),
		Transparency = 1,
		CanQuery = false,
		Parent = model,
	})
	model.PrimaryPart = pivot

	local stemHeight = size * 0.35
	ModelFactory.Part({
		Name = "Stem",
		Size = Vector3.new(0.3, stemHeight, 0.3),
		CFrame = CFrame.new(0, stemHeight / 2, 0),
		Color = Palette.Stem,
		Parent = model,
	})
	for index = 0, 2 do
		wedge({
			Name = "Leaf",
			Size = Vector3.new(0.1, size * 0.25, size * 0.45),
			CFrame = CFrame.Angles(0, math.rad(index * 120), 0) * CFrame.new(0, size * 0.12, -size * 0.25),
			Color = Palette.Stem,
			Parent = model,
		})
	end

	local fruitParent: BasePart? = nil
	local kind = visual.Kind
	if scale >= 0.4 then
		if kind == "Star" then
			fruitParent = ModelFactory.Part({
				Name = "Fruit",
				Shape = Enum.PartType.Ball,
				Size = Vector3.one * size * 0.6,
				CFrame = CFrame.new(0, stemHeight + size * 0.3, 0),
				Color = visual.Color,
				Material = Enum.Material.Neon,
				Parent = model,
			})
			for index = 0, 4 do
				local angle = math.rad(index * 72)
				ModelFactory.Part({
					Name = "Spike",
					Shape = Enum.PartType.Ball,
					Size = Vector3.one * size * 0.22,
					CFrame = CFrame.new(0, stemHeight + size * 0.3, 0)
						* CFrame.Angles(0, angle, 0)
						* CFrame.new(0, 0, -size * 0.36),
					Color = visual.SecondaryColor,
					Material = Enum.Material.Neon,
					Parent = model,
				})
			end
		elseif kind == "Pepper" then
			fruitParent = ellipsoid({
				Name = "Fruit",
				Size = Vector3.new(size * 0.35, size * 0.8, size * 0.35),
				CFrame = CFrame.new(0, stemHeight + size * 0.3, 0) * CFrame.Angles(0, 0, math.rad(15)),
				Color = visual.Color,
				Material = Enum.Material.Glass,
				Transparency = 0.15,
				Parent = model,
			})
		elseif kind == "Corn" then
			ModelFactory.Part({
				Name = "Stalk",
				Size = Vector3.new(0.4, size, 0.4),
				CFrame = CFrame.new(0, size / 2, 0),
				Color = visual.SecondaryColor,
				Parent = model,
			})
			fruitParent = ellipsoid({
				Name = "Fruit",
				Size = Vector3.new(size * 0.25, size * 0.5, size * 0.25),
				CFrame = CFrame.new(size * 0.18, size * 0.6, 0),
				Color = visual.Color,
				Material = Enum.Material.Glass,
				Transparency = 0.3,
				Parent = model,
			})
		elseif kind == "Pumpkin" or kind == "Melon" then
			local center = CFrame.new(0, size * 0.4, 0)
			fruitParent = ellipsoid({
				Name = "Fruit",
				Size = Vector3.new(size, size * 0.8, size),
				CFrame = center,
				Color = visual.Color,
				Parent = model,
			})
			local ribs = if kind == "Pumpkin" then 4 else 3
			for index = 0, ribs - 1 do
				ellipsoid({
					Name = "Rib",
					Size = Vector3.new(size * 0.45, size * 0.82, size * 1.02),
					CFrame = center * CFrame.Angles(0, math.rad(index * (180 / ribs)), 0),
					Color = if kind == "Pumpkin" then visual.Color else visual.SecondaryColor,
					Parent = model,
				})
			end
			ModelFactory.Part({
				Name = "FruitStem",
				Size = Vector3.new(0.4, size * 0.25, 0.4),
				CFrame = center * CFrame.new(0, size * 0.45, 0),
				Color = Palette.Stem,
				Parent = model,
			})
		end
	end

	if fruitParent and visual.Glow and scale >= 1 then
		local light = Instance.new("PointLight")
		light.Color = visual.SecondaryColor
		light.Range = 10
		light.Brightness = 1.5
		light.Parent = fruitParent
	end

	for _, d in model:GetDescendants() do
		if d:IsA("BasePart") then
			d.CanCollide = false
			d.CanTouch = false
		end
	end
	return model
end
```

```lua
-- Returns the camera ray under the pointer. Touch positions are GUI coordinates

	local camera = Workspace.CurrentCamera

local function groundPointAt(ray: Ray): Vector3?
	local folder = ClientState.MatchFolder

	if ground == nil then
		return nil

	params.FilterDescendantsInstances = { ground }
	local result = Workspace:Raycast(ray.Origin, ray.Direction * 300, params)

local function looksValid(position: Vector3): boolean
	local folder = ClientState.MatchFolder

local function bindDefense(model: Instance)
	if not model:IsA("Model") then
		return
	end
	local defenseId = model:GetAttribute(Attributes.DefenseId)
	local definition = defenseId and DefenseConfig.Defenses[defenseId]
	local base = model.PrimaryPart or model:WaitForChild("Base", 5)
	if definition == nil or base == nil or not base:IsA("BasePart") then
		return
	end
	local prompt = Instance.new("ProximityPrompt")
	prompt.Name = "UpgradePrompt"
	prompt.KeyboardKeyCode = Enum.KeyCode.F
	prompt.GamepadKeyCode = Enum.KeyCode.ButtonX
	prompt.HoldDuration = 0.5
	prompt.RequiresLineOfSight = false
	prompt.MaxActivationDistance = 12
	prompt.Parent = base
	local function update()
		local level = model:GetAttribute(Attributes.Level) or 0
		local upgrade = definition.Upgrades and definition.Upgrades[level + 1]
		local health = model:GetAttribute(Attributes.Health) or 0
		local maxHealth = model:GetAttribute(Attributes.MaxHealth) or 1
		prompt.ObjectText = `{definition.DisplayName} Lv{level} • {Format.Commas(health)}/{Format.Commas(maxHealth)}`
		prompt.ActionText = if upgrade then `Upgrade 🪙 {upgrade.Cost}` else "Max level"
		prompt.Enabled = upgrade ~= nil
	end
	model:GetAttributeChangedSignal(Attributes.Level):Connect(update)
	model:GetAttributeChangedSignal(Attributes.Health):Connect(update)
	update()
	prompt.Triggered:Connect(function()
		task.spawn(function()
			local result = Net:Invoke("RequestUpgradeDefense", model:GetAttribute(Attributes.DefenseUid))
			Controllers.NotificationController:ShowResult(result, "Upgraded!")
		end)
	end)
end

local function bindMatch(folder: Folder?)
	matchJanitor:Cleanup()
	table.clear(ownedAccessories)
	DefenseShopController:CancelPlacement()
	if folder == nil then
		Panels.Close("DefenseUI")
		return
	end

	if defenses then
		matchJanitor:Connect(defenses.ChildAdded, function(child)
			task.spawn(bindDefense, child)
		end)
		for _, child in defenses:GetChildren() do
			task.spawn(bindDefense, child)
		end
	end
end
```

```lua
local function distanceToDoor(extraction: any): number
	local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart") :: BasePart?
	if root == nil or extraction.DoorPosition == nil then
		return math.huge
	end
	return (root.Position - extraction.DoorPosition).Magnitude
end

function EscapeController:Decide(remote: string)
	local extraction = ClientState.Extraction
	if deciding or extraction == nil or not extraction.Open or extraction.YourDecision ~= nil then
		return
	end
	if distanceToDoor(extraction) > MatchConfig.Interaction.DoorRange then
		Controllers.NotificationController:Notify("Walk to the Escape Door first!", "Warning", 2)
		return
	end
	deciding = true
	task.spawn(function()
		local result = Net:Invoke(remote)
		deciding = false
		Controllers.NotificationController:ShowResult(result, if remote == "RequestEscape" then nil else "You chose to CONTINUE. Brace yourself!")
	end)
end

local function refresh()
	local extraction = ClientState.Extraction
	local open = extraction ~= nil and extraction.Open == true and ClientState.InMatch()
	ui.Frame.Visible = open
	if not open then
		return
	end
	ui.Escape.Text = `ESCAPE TO THE LOBBY\n<font size="18">Bank {Format.Commas(extraction.EstimatedCandies or 0)} 🍬</font>`
	ui.Continue.Text = `CONTINUE\n<font size="18">Danger {Format.Roman(extraction.NextContinueLevel or 1)} • zombies {Format.Multiplier(extraction.NextHealthMultiplier or 1)}</font>`

	local names = {}
	for userIdText, decision in extraction.Decisions or {} do
		local other = Players:GetPlayerByUserId(tonumber(userIdText) or 0)
		table.insert(names, `{other and other.DisplayName or "?"}: {decision}`)
	end
	local mine = extraction.YourDecision
	local near = distanceToDoor(extraction) <= MatchConfig.Interaction.DoorRange
	local locked = mine ~= nil
	ui.Escape.Active = near and not locked
	ui.Continue.Active = near and not locked
	ui.Escape.AutoButtonColor = false
	ui.Escape.BackgroundTransparency = if near and not locked then 0 else 0.5
```
